out={}
for ins,ns in [('600058.SH',range(248,256)),('600089.SH',range(345,348))]:
 pages=json.loads((P/f'quote_v47_unseen_source_{ins}.json').read_text());layouts=json.loads((P/f'quote_v47_layout_{ins}.json').read_text());ledger=[]

   ms=list(num.finditer(line))
   if not ms:
    if line.strip():buf.append(line.strip())
    continue
   vals=tuple(m.group() for m in ms)
   assert lrows[idx][0]==vals,(ins,n,idx,lrows[idx],vals)
   current=lrows[idx][1];idx+=1
   native=co(''.join(buf)+line[:ms[0].start()]);buf=[]

   # Original plain cell order supplies complete wrapped names; layout fixes annual placement.

   assert party and content,(n,native)
   ledger.append(dict(page=n,kind='transaction',counterparty=party,content=content,current_value=vals[0] if current else None,prior_value=(vals[1] if len(vals)>1 else vals[0] if not current else None),native= native,layout_current_end=current_end,layout_prior_end=prior_end))

 transactions=[x for x in ledger if x['kind']=='transaction'];curr=[x for x in transactions if x['current_value'] is not None];prior=[x for x in transactions if x['current_value'] is None]
