def read(path):
    return json.loads(Path(path).read_text())

def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()

def dump(path, value):
    Path(path).parent.mkdir(parents=True, exist_ok=True)
    Path(path).write_text(json.dumps(value, ensure_ascii=False, indent=2, default=str) + '\n')

class Capture(H['ContentCapture']):
    """Task-local safe wire timing; gateway retains retries/schema/usage ownership."""
    def __init__(self, delegate, directory):
        super().__init__(delegate, directory)
        self.wires = []

    async def send(self, url, headers, payload, timeout_seconds):
        maximum = 1200 if payload.get('model') == 'grok-4.7' else 600
        assert 0 < timeout_seconds <= maximum + 0.01
        encoded = json.dumps(payload, ensure_ascii=False, sort_keys=True).encode()
        entry = {'model': payload.get('model'), 'physical_timeout_seconds': timeout_seconds,
            'payload_bytes': len(encoded), 'payload_sha256': hashlib.sha256(encoded).hexdigest(),
            'max_tokens': payload.get('max_tokens'), 'reasoning_effort': payload.get('reasoning_effort'),
            'headers_or_material_recorded': False, 'state': 'sending'}
        self.wires.append(entry)
        dump(D / 'request-wiring.json', self.wires)
        started = time.monotonic()
        try:
            response = await super().send(url, headers, payload, timeout_seconds)
            entry.update(state='returned', status_code=response.status_code)
            return response
        except Exception as exc:
            entry.update(state='transport_error', exception_type=type(exc).__name__)
            raise
        finally:
            entry['seconds'] = round(time.monotonic() - started, 3)
            dump(D / 'request-wiring.json', self.wires)
