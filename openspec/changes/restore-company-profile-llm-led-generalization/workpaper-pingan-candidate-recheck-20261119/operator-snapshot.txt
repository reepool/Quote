def read(path): return json.loads(Path(path).read_text())
def sha(path): return hashlib.sha256(Path(path).read_bytes()).hexdigest()
def digest(value): return hashlib.sha256(json.dumps(value, ensure_ascii=False, sort_keys=True).encode()).hexdigest()
def dump(path, value):
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2, default=str) + '\n')

def materials():
    from research.company_profile.workpaper import CompanyProfileWorkpaper, workpaper_element_payloads
    from research.company_profile.workpaper_review import workpaper_hash
    paper = CompanyProfileWorkpaper.model_validate(read(OLD / '601318/runtime.json')['workpaper'])
    payload = paper.model_dump(mode='json')
    payload['quality_review'] = None
    payload.pop('use_review', None)
    payload['standardization_notes'] = [n for n in payload['standardization_notes']
        if not n.startswith(('复核目标标准化：', '修订源检查隔离：'))]
    originals = {f.affected_fact_id: f.original_fact.model_dump(mode='json')
                 for f in paper.quality_review.findings if f.original_fact is not None}
    for element in workpaper_element_payloads(payload).values():
        element['facts'] = [originals.get(f['fact_id'], f) for f in element['facts']]
    seed = CompanyProfileWorkpaper.model_validate(payload)
    assert workpaper_hash(seed) == paper.quality_review.original_draft_hash
    receipts = [r for r in read(OLD / '601318/model-receipts.json')
                if r['purpose'] in {'review_profile_workpaper', 'repair_profile_workpaper'}]
    assert len(receipts) == 2
    resume = dict(original_draft_hash=workpaper_hash(seed), document_version=paper.report.document_version,
        review_response=next(r['response_payload'] for r in receipts if r['purpose'] == 'review_profile_workpaper'),
        repair_response=next(r['response_payload'] for r in receipts if r['purpose'] == 'repair_profile_workpaper'), receipts=receipts)
    item = read(OLD / 'source-manifest.json')['companies'][0]
    pdf = Path(item['asset']['archive_path'])
    assert sha(pdf) == paper.report.document_version == item['asset']['content_hash']
    cache = pdf.parent / 'derived/business_profile_pdfium_recovery.v4' / (pdf.stem + '_57e1dd4f16ef40ff.json.gz')
    pages = json.loads(gzip.decompress(cache.read_bytes()))['pages']
    return seed, resume, item, pages
