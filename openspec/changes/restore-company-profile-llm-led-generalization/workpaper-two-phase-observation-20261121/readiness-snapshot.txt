ROOT = Path('/home/python/Quote')
sys.path.insert(0, str(ROOT))
TEMP = Path(__file__).parent
LOCAL = TEMP / 'ready-final'
CHANGE = ROOT / 'openspec/changes/restore-company-profile-llm-led-generalization'
SOURCE = CHANGE / 'workpaper-s2-confirmation-result-20261119'
EXTRACT, REVIEW = 'semantic_extraction__deepseek', 'semantic_extraction__scorpio_grok'
ORDER = ('002142.SZ', '920002.BJ')


def read(path):
    return json.loads(Path(path).read_text())


def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def dump(path, value):
    Path(path).write_text(json.dumps(value, ensure_ascii=False, indent=2) + '\n')


def task_config():
    from utils.llm.config import load_project_environment
    from utils.llm.models import LlmConfig
    load_project_environment(ROOT)
    original = LlmConfig.from_mapping(read(ROOT / 'config/13_llm.json')['llm'])
    assert original.enabled
    profiles = {}
    for name, host, model, wait in ((EXTRACT, 'api.deepseek.com', 'deepseek-flash', 600),
                                   (REVIEW, 'scorpio.reepool.com', 'grok-4.7', 1200)):
        profile = original.profiles[name]
        assert profile.enabled and os.environ.get(profile.api_key_env)
        assert urlsplit(profile.base_url).hostname == host and profile.model == model
        profiles[name] = replace(profile, timeout_seconds=2450, attempt_timeout_seconds=wait,
            max_retries=1, max_schema_repair_attempts=1)
    assert profiles[REVIEW].reasoning_effort == 'high'
    profiles[EXTRACT] = replace(profiles[EXTRACT], max_output_tokens_field='max_tokens', disable_thinking=True)
    return replace(original, profiles=profiles, routes={}, pools={})


def load_sources():
    from research.company_profile.models import ReportIdentity
    sources, metadata = {}, []
    for instrument in ORDER:
        symbol = instrument[:6]
        path, binding = SOURCE / 'source-bindings' / (symbol + '.json'), SOURCE / symbol / 'binding.json'
        source, old_binding = read(path), read(binding)
        asset = source['asset']
        cache = Path(old_binding['source_preparation']['artifact_path'])
        assert asset['instrument_id'] == instrument and asset['report_period'] == '2025-12-31'
        assert sha(asset['archive_path']) == asset['content_hash']
        with gzip.open(cache, 'rt', encoding='utf-8') as stream:
            cached = json.load(stream)
        assert cached['source_content_hash'] == asset['content_hash']
        pages = [{'page': p['page_number'], 'text': p['text']} for p in cached['pages']]
        assert len(pages) == old_binding['physical_pages']
        report = ReportIdentity(instrument_id=instrument, report_id=asset['filing_id'],
            document_version=asset['content_hash'], report_period=asset['report_period'],
            published_at=asset['published_at'], document_type='annual_report')
        sources[instrument] = dict(source=source, report=report, pages=pages)
        metadata.append(dict(instrument_id=instrument, report=report.model_dump(mode='json'),
            source_binding_json=str(path.relative_to(ROOT)), source_binding_sha256=sha(path),
            native_cache_path=str(cache), native_cache_file_sha256=sha(cache), physical_pages=len(pages),
            old_semantic_or_navigation_loaded=False))
    return sources, metadata


def prepare():
    from research.announcements import AnnouncementRecord, AnnouncementAttachment
    from research.business_profile_production_operations import BusinessProfileAnnouncementFrontierRepository
    from research.company_profile.operations import execute_published_task
    from research.company_profile.stage5_provider import CommonGatewaySemanticProvider
    from research.company_profile.workpaper_llm import workpaper_processing_identity
    from research.storage import ResearchStorageManager
    from utils.config_manager import ResearchConfig, ResearchStorageConfig
    from utils.llm.budget import LlmRunBudget, LogicalRecoveryPolicy, DEVELOPMENT_UNPRICED_BILLING
    from utils.llm.client import LlmClient
    from utils.llm.testing import ScriptedTransport
    from utils.logging_manager import LoggingManager, LogConfig
    started = time.monotonic()
    LOCAL.mkdir()
    LoggingManager().configure(LogConfig(level='WARNING', enable_console=False,
        enable_file=True, log_directory=str(LOCAL / 'logs')))
    assert os.environ['TMPDIR'] == str(ROOT / 'data/QuoteBak')
    sources, items = load_sources()
    config = task_config()
    # Real profiles, but a transport that cannot reach any external host.
    transport = ScriptedTransport([])
    local_budget = LlmRunBudget(deadline=time.monotonic() + 12600, prices={}, max_tokens=2000000,
        max_requests=20, max_cost_cny=Decimal(20), billing_mode=DEVELOPMENT_UNPRICED_BILLING,
        recovery_policy=LogicalRecoveryPolicy(6, 1), run_authorization='company_profile_workpaper_development.2026-11-11')
    client = LlmClient(config, transport=transport, physical_budget=local_budget)
    provider = CommonGatewaySemanticProvider(client=client, profile=EXTRACT, max_output_tokens=12000,
        timeout_seconds=2450, execution_authorization={'status': 'not_authorized'})
    identity = workpaper_processing_identity(real=True, review=True, review_profile=REVIEW)
    identity.update(logical_request_timeout_seconds=2450, review_max_output_tokens=16384, repair_profile=EXTRACT)
    db = str(LOCAL / 'research.db')
    storage = ResearchStorageManager(ResearchConfig(storage=ResearchStorageConfig(db_path=db,
        financials_db_path=db, valuation_db_path=db, interests_db_path=db,
        quotes_db_path=str(LOCAL / 'quotes.db'), attach_quotes_db=False)))
    storage.initialize()
    frontier = BusinessProfileAnnouncementFrontierRepository(storage)
    for instrument, data in sources.items():
        asset, announcement = data['source']['asset'], data['source']['announcement']
        exchange = 'SZSE' if instrument.endswith('.SZ') else 'BSE'
        frontier.upsert_record(instrument={'instrument_id': instrument, 'symbol': instrument[:6], 'exchange': exchange},
            record=AnnouncementRecord(source='cninfo', source_announcement_id=asset['filing_id'],
                announcement_key='cninfo:' + asset['filing_id'], title=announcement['title'],
                published_at=asset['published_at'], exchange=exchange, symbols=(instrument[:6],),
                attachments=(AnnouncementAttachment(source_url=announcement['source_url'],
                    resolved_url=announcement['source_url'], file_extension='PDF'),)))
    captures = []
    actual_identities = {}

    class BeforeSend(BaseException):
        pass

    def stop_before_request(**kwargs):
        assert kwargs['purpose'] == 'select_profile_sections'
        payload = kwargs['payload']
        actual_identity = actual_identities[kwargs['report'].instrument_id]
        assert actual_identity['workpaper_review_execution'] == 'deferred'
        captures.append(dict(instrument_id=kwargs['report'].instrument_id, purpose=kwargs['purpose'],
            actual_first_read_identity=actual_identity, payload_keys=list(payload),
            payload_bytes=len(json.dumps(payload, ensure_ascii=False, sort_keys=True).encode()),
            payload_sha256=hashlib.sha256(json.dumps(payload, ensure_ascii=False, sort_keys=True).encode()).hexdigest(),
            schema_sha256=hashlib.sha256(json.dumps(kwargs['response_model'].model_json_schema(), sort_keys=True).encode()).hexdigest(),
            max_output_tokens=kwargs['max_output_tokens'], timeout_seconds=kwargs['timeout_seconds']))
        # Keep the input locally only for human inspection; never persist full duplicate source.
        dump(TEMP / (kwargs['report'].instrument_id[:6] + '-selection-input.json'), payload)
        raise BeforeSend()

    provider.workpaper_request = stop_before_request

    def source_for(item):
        from research.company_profile.execution import processing_identity_from_item
        actual_identities[item['instrument_id']] = processing_identity_from_item(item)
        data = sources[item['instrument_id']]
        return dict(report=data['report'], pages=data['pages'], source_pdf_path=data['source']['asset']['archive_path'])

    try:
        for instrument in ORDER:
            try:
                asyncio.run(execute_published_task(action='first_read', storage=storage,
                    output_root=LOCAL / 'output', checkpoint_root=LOCAL / instrument[:6], page_source=source_for,
                    provider=provider, processing_identity=identity, execution_budget=local_budget,
                    knowledge_cutoff='2026-11-21', instrument_ids=[instrument], max_items=1,
                    token_budget=2000000, max_elapsed_seconds=120))
            except BeforeSend:
                pass
        assert [c['instrument_id'] for c in captures] == list(ORDER)
        assert not transport.calls and not local_budget.snapshot()['physical_requests']
    finally:
        asyncio.run(client.close())
    protected = read(CHANGE / 'workpaper-batch-query-20261121/saved-batch-verification.json')['protected_20_writer_sha256']
    assert all(sha(ROOT / path) == digest for path, digest in protected.items())
    result = dict(task='6.EP', status='locally_ready_not_authorized', instruments=list(ORDER), items=items,
        prepare_entry_captures=captures, identity=identity, provider_constructor_timeout_seconds=provider._timeout_seconds,
        models={name: dict(model=p.model, host=urlsplit(p.base_url).hostname,
            reasoning_effort=p.reasoning_effort, timeout_seconds=p.timeout_seconds,
            attempt_timeout_seconds=p.attempt_timeout_seconds, max_retries=p.max_retries,
            max_schema_repair_attempts=p.max_schema_repair_attempts) for name, p in config.profiles.items()},
        shared_config_sha256=sha(ROOT / 'config/13_llm.json'),
        code_sha256={str(p.relative_to(ROOT)):sha(p) for p in sorted((ROOT/'research/company_profile').glob('*.py'))
            + sorted((ROOT/'utils/llm').glob('*.py'))},
        reused_validation='workpaper-task-entry-20261121/validation.json',
        protected_writer_hash_reference='workpaper-batch-query-20261121/saved-batch-verification.json',
        protected_writers=20, protected_writers_unchanged=True,
        local_prepare_seconds=round(time.monotonic()-started,3), new_model_requests=0,
        new_data_transfers=0, formal_attempt_started=False, production_authorization='not_authorized')
    dump(TEMP/'readiness.json', result)
    print(json.dumps(dict(status=result['status'], companies=len(items), captures=len(captures),
        pages=[i['physical_pages'] for i in items], new_model_requests=0,
        local_prepare_seconds=result['local_prepare_seconds']), ensure_ascii=False), flush=True)


if __name__ == '__main__':
    prepare()
