def read(path):
    return json.loads(Path(path).read_text())


def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def dump(path, value):
    Path(path).parent.mkdir(parents=True, exist_ok=True)
    Path(path).write_text(json.dumps(value, ensure_ascii=False, indent=2, default=str) + '\n')


class Capture(H['ContentCapture']):
    def __init__(self, delegate, directory):
        super().__init__(delegate, directory)
        self.wires = []

    async def send(self, url, headers, payload, timeout_seconds):
        assert urlsplit(url).hostname == 'scorpio.reepool.com'
        assert payload.get('model') == 'grok-4.7' and payload.get('reasoning_effort') == 'high'
        assert 0 < timeout_seconds <= 1200.01 and len(self.wires) < 2
        encoded = json.dumps(payload, ensure_ascii=False, sort_keys=True).encode()
        wire = {'model': payload['model'], 'reasoning_effort': payload['reasoning_effort'],
                'physical_timeout_seconds': timeout_seconds, 'payload_bytes': len(encoded),
                'payload_sha256': hashlib.sha256(encoded).hexdigest(), 'max_tokens': payload.get('max_tokens'),
                'state': 'sending', 'headers_or_private_material_recorded': False}
        self.wires.append(wire)
        dump(D / 'request-wiring.json', self.wires)
        started = time.monotonic()
        try:
            response = await super().send(url, headers, payload, timeout_seconds)
            wire.update(state='returned', status_code=response.status_code)
            return response
        except Exception as exc:
            wire.update(state='transport_error', exception_type=type(exc).__name__)
            raise
        finally:
            wire['seconds'] = round(time.monotonic() - started, 3)
            dump(D / 'request-wiring.json', self.wires)
