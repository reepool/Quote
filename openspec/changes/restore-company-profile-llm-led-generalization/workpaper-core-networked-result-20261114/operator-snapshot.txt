def dump(path, value):
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2, default=str) + '\n')

def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def inspect(directory, symbol, pages=None, sections=None):
    prepared=json.loads((directory/(symbol+'.json')).read_text())
    if sections:
        for entry in prepared['index']:
            if re.search(sections,entry['section_path'][-1]):
                print(json.dumps({k:entry[k] for k in ('section_id','section_path','start_page','end_page','characters')},ensure_ascii=False))
        return
    from research.company_profile.workpaper_sections import extract_selected_layout
    page_numbers=[int(v) for v in pages.split(',')]
    for page in extract_selected_layout(Path(prepared['asset']['archive_path']).read_bytes(), page_numbers):
        print('\nPHYSICAL PAGE',symbol,page['page'])
        print(page['text'])

def network_preflight():
    import socket,ssl
    from urllib.parse import urlsplit
    config=config_for_run()
    assert urlsplit(config.profiles[EXTRACT].base_url).hostname=='api.deepseek.com'
    checks=[]
    for host in ('api.deepseek.com','scorpio.reepool.com'):
        begun=time.monotonic()
        found=socket.getaddrinfo(host,443,type=socket.SOCK_STREAM)
        with socket.create_connection((host,443),timeout=10) as sock:
            with ssl.create_default_context().wrap_socket(sock,server_hostname=host) as tls:
                checks.append({'host':host,'dns_addresses':len(found),'tls':tls.version(),
                    'seconds':round(time.monotonic()-begun,3)})
    context={'task':'6.CG','namespace':os.readlink('/proc/self/ns/net'),
        'permission_mode':'approved_network_exec','checks':checks,'model_requests':0}
    dump(DEST/'network-preflight.json',context)
    print(json.dumps(context,ensure_ascii=False),flush=True)

def group(directory,batch):
    assert batch in (1,2,3)
    frozen=json.loads((DEST/'authorization.json').read_text())
    assert code_hashes()==frozen['code_sha256']
    assert sha(Path(__file__))==frozen['operator_sha256']
    assert not (DEST/'independent-stop.json').exists()
    assert not (DEST/'group-stop.json').exists()
    state=DEST/'group-clock.json'
    if batch==1:
        assert not state.exists()
        started=time.monotonic()
        dump(state,{'started_monotonic':started,'deadline_monotonic':started+7200})
    else:
        prior=json.loads((DEST/('batch-'+str(batch-1)+'-execution.json')).read_text())
        assert prior['executed']==4 and prior['stop_reason'] is None
        assert (DEST/('batch-'+str(batch-1)+'-source-review.json')).exists()
    clock=json.loads(state.read_text()); deadline=clock['deadline_monotonic']
    assert time.monotonic()<deadline
    items=source_items()[(batch-1)*4:batch*4]
    processes={}; logs={}; results={}; stop=None
    for item in items:
        symbol=item['instrument_id'][:6]
        assert not (DEST/symbol).exists()
        log=directory/(symbol+'-run.log'); logs[symbol]=log.open('x')
        processes[symbol]=subprocess.Popen([sys.executable,str(Path(__file__)),'run',
            '--prepared',str(directory),'--symbol',symbol,'--group-deadline',str(deadline)],
            stdout=logs[symbol],stderr=subprocess.STDOUT)
    while len(results)<len(items):
        for symbol,process in processes.items():
            timing_path=DEST/symbol/'entry-timing.json'
            if process.poll() is None and timing_path.exists():
                timing=json.loads(timing_path.read_text())
                if time.monotonic()>=timing['absolute_deadline_monotonic']:
                    process.send_signal(signal.SIGUSR1)
                    process.terminate()
                    dump(DEST/symbol/'supervisor-hard-stop.json',{'reason':'absolute_company_deadline','monotonic':time.monotonic()})
        for symbol,process in processes.items():
            if symbol in results or process.poll() is None:continue
            logs[symbol].close()
            result_path=DEST/symbol/'summary.json'
            results[symbol]=json.loads(result_path.read_text()) if result_path.exists() else {
                'instrument_id':symbol+'.SH','exception':'operator_exit_'+str(process.returncode),'readable':False}
            print('COMPANY_DONE '+json.dumps(results[symbol],ensure_ascii=False),flush=True)
        all_results={}
        for item in source_items():
            symbol=item['instrument_id'][:6]
            path=DEST/symbol/'summary.json'
            if path.exists():all_results[symbol]=json.loads(path.read_text())
        ordered=[all_results.get(i['instrument_id'][:6]) for i in source_items()]
        if stop is None:
            if time.monotonic()>=deadline:stop='group_deadline'
            else:
                for left,right in zip(ordered,ordered[1:]):
                    if left is None or right is None:continue
                    x=left.get('failure_signature') or (left.get('exception') if not left.get('readable') else None)
                    y=right.get('failure_signature') or (right.get('exception') if not right.get('readable') else None)
                    if x and x==y:
                        stop='consecutive_same_core_failure: '+x
                        break
            if (DEST/'independent-stop.json').exists():stop='independent_source_stop'
            if stop:
                dump(DEST/'group-stop.json',{'reason':stop,'batch':batch,'at_monotonic':time.monotonic()})
                for process in processes.values():
                    if process.poll() is None:process.send_signal(signal.SIGUSR1)
        dump(DEST/'progress.json',{'batch':batch,'executed':len(all_results),
            'results':[v for v in ordered if v],'stop_reason':stop,
            'supervision_wall_seconds':round(time.monotonic()-clock['started_monotonic'],3)})
        if len(results)<len(items):time.sleep(1)
    dump(DEST/('batch-'+str(batch)+'-execution.json'),{'executed':len(results),
        'results':[results[i['instrument_id'][:6]] for i in items],'stop_reason':stop,
        'source_code_unchanged':code_hashes()==frozen['code_sha256']})
    print('BATCH_DONE '+str(batch)+' '+str(stop),flush=True)

def main():
    parser=argparse.ArgumentParser()
    parser.add_argument('mode',choices=['prepare','freeze','run','group','preflight'])
    parser.add_argument('--prepared',type=Path)
    parser.add_argument('--symbol')
    parser.add_argument('--batch',type=int)
    parser.add_argument('--group-deadline',type=float)
    args=parser.parse_args()
    if args.mode=='prepare':prepare()
    elif args.mode=='freeze':freeze(args.prepared)
    elif args.mode=='run':run_one(args.prepared,args.symbol,args.group_deadline)
    elif args.mode=='preflight':network_preflight()
    else:group(args.prepared,args.batch)
if __name__=='__main__':main()
