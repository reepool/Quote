def read(path):
    return json.loads(path.read_text())


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def dump(path, value):
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2) + '\n')


def freeze_results():
    authorization = read(DEST / 'authorization.json')
    assert all(sha(ROOT / p) == h for p, h in authorization['code_sha256'].items())
    assert sha(ROOT / 'config/13_llm.json') == authorization['shared_config_sha256']
    assert sha(DEST / 'operator-snapshot.txt') == authorization['operator_sha256']
    rows, providers = [], Counter()
    for symbol in SYMBOLS:
        directory = DEST / symbol
        summary, budget = read(directory / 'summary.json'), read(directory / 'budget.json')
        assert summary['three_reads_equal']
        assert summary['physical_requests'] == budget['physical_requests'] <= 10
        assert summary['charged_tokens'] == budget['charged_tokens'] <= 500000
        assert summary['business_seconds'] <= 1800
        assert summary['recoveries'] <= 3
        providers.update(a['source_label'] for a in budget['attempts'])
        rows.append({k: summary.get(k) for k in ('instrument_id', 'readable', 'physical_requests', 'charged_tokens', 'recoveries', 'business_seconds', 'supervision_seconds', 'preparation_seconds', 'cleanup_bytes', 'review_status', 'review_error', 'exception')})
    assert sum(r['physical_requests'] for r in rows) == 13
    assert sum(r['charged_tokens'] for r in rows) == 365532
    original_hashes = {str(p.relative_to(DEST)): sha(p) for p in sorted(DEST.rglob('*')) if p.is_file()}
    dump(DEST / 'original-result-check.json', {
        'task': '6.CK', 'checked_before_post_run_code_changes': True,
        'frozen_source_and_configuration_equal': True,
        'original_file_sha256': original_hashes, 'companies': rows,
        'physical_requests': 13, 'charged_tokens': 365532, 'recoveries': 1,
        'providers': dict(providers), 'cost': 'unknown',
        'readable_profiles': 3, 'completed_review_repair_loops': 0,
        's1': '0/12', 'production_authorization': 'not_authorized',
    })
    print(json.dumps({'frozen_originals': len(original_hashes), 'providers': dict(providers), 'companies': rows}, ensure_ascii=False))


def local_checks():

    from research.company_profile.workpaper_review import workpaper_hash
    from research.company_profile.workpaper_sections import SectionSelection, IndexedSection, index_sections, selected_pages, extract_selected_layout, build_section_packet
    from research.company_profile.runtime import CompanyProfileStageRuntime, CompanyProfileResearchWriter
    from research.company_profile.reads import CompanyProfileReadService

    original = read(DEST / 'original-result-check.json')
    assert all(sha(DEST / p) == h for p, h in original['original_file_sha256'].items())
    prepared = read(PREPARED / '600054.json')
    report = ReportIdentity.model_validate(read(DEST / '600082/query.json')['profiles'][0]['report']).model_copy(update={
        'instrument_id': '600054.SH', 'document_version': prepared['asset']['content_hash'],
        'source_asset_id': prepared['asset']['source_asset_id'],
    })
    # This maps unchanged titles, NOT the new index IDs onto the old selection.
    old = [IndexedSection.model_validate(i) for i in prepared['index']]
    selection_raw = read(DEST / '600054/model-receipts.json')[0]['response_payload']
    old_selection = SectionSelection.model_validate(selection_raw)
    selected_paths = [s.section_path for s in old if s.section_id in old_selection.selected_section_ids]
    fresh = index_sections(prepared['pages'])
    mapping = {}
    for path in selected_paths:
        matches = [s for s in fresh if s.section_path == path]
        assert len(matches) == 1, (path, len(matches))
        mapping[path] = matches[0]
    selection = old_selection.model_copy(update={'selected_section_ids': tuple(s.section_id for s in mapping.values())})
    numbers = selected_pages(fresh, selection)
    pdf = Path(prepared['asset']['archive_path'])
    assert sha(pdf) == prepared['asset']['content_hash']
    layout = extract_selected_layout(pdf.read_bytes(), numbers)
    packet = build_section_packet(report, layout, fresh, selection, remap_layout=True, max_characters=500000)
    assert '000.00 自有资金' not in [s.section_path[-1] for s in fresh]
    check = {
        'execution': 'post_run_local_source_packet_only', 'new_model_requests': 0,
        'original_failure_retained': True, 'preprocessing_version': workpaper_processing_identity(real=False)['preprocessing_version'],
        'source_sha256': sha(pdf), 'selected_title_paths_unique': True,
        'selection_mapping': [{'path': list(path), 'old_id': next(s.section_id for s in old if s.section_path == path), 'new_id': s.section_id} for path, s in mapping.items()],
        'packet_pages': sorted({sp.page for e in packet.evidence for sp in e.source_spans}),
        'packet_characters': sum(len(sp.quote) for e in packet.evidence for sp in e.source_spans),
        'all_packet_coordinates_equal': all(next(p['text'] for p in layout if p['page'] == sp.page)[sp.start:sp.end] == sp.quote for e in packet.evidence for sp in e.source_spans),
        'not_a_new_huangshan_profile': True,
    }
    dump(DEST / 'huangshan-local-source-check.json', check)

    receipts = read(DEST / '600082/model-receipts.json')
    saved = read(DEST / '600082/query.json')['profiles'][0]['workpaper']
    paper = CompanyProfileWorkpaper.model_validate(saved)
    extraction = next(r for r in receipts if r['purpose'] == 'extract_profile_workpaper')
    initial = next(r for r in receipts if r['purpose'] == 'review_profile_workpaper')
    seed = bind_workpaper_draft(WorkpaperDraftResponse.model_validate(extraction['response_payload']), report=paper.report,
        reading_scope=paper.reading_scope, evidence=paper.evidence, origin='llm_generated', generation=paper.generation)
    assert workpaper_hash(seed) == paper.quality_review.original_draft_hash
    seed = seed.model_copy(update={'origin': 'controlled_fixture'})
    class SavedReviewOnly:
        offline_only = True
        workpaper_receipts = ()
        def __init__(self):
            self.calls = []
        def workpaper_request(self, **kwargs):
            self.calls.append(kwargs)
            if kwargs['purpose'] == 'review_profile_workpaper':
                return initial['response_payload']
            raise ValueError('offline_stop_no_new_repair_or_network_authorized')

    provider = SavedReviewOnly()
    source_pdf = Path(read(PREPARED / '600082.json')['asset']['archive_path'])
    seed_pages = extract_selected_layout(source_pdf.read_bytes(), sorted({sp.page for ev in seed.evidence for sp in ev.source_spans}))
    root = Path(tempfile.mkdtemp(prefix='workpaper-ck-local-readback-'))
    destination = DEST / 'local-review-replay'
    destination.mkdir(exist_ok=True)
    try:
        runtime = CompanyProfileStageRuntime(writer=CompanyProfileResearchWriter(root), provider=provider)
        identity = workpaper_processing_identity(real=False, review=True)
        identity['review_existing_draft_hash'] = workpaper_hash(seed)
        item = {'work_id': 'ck-local-review-replay', 'report': seed.report, 'review_seed': seed.model_dump(mode='json'),
                'pages': seed_pages,
                'source_pdf_path': str(source_pdf), 'processing_identity': identity}
        for stage in ('acquire', 'parse', 'semantic', 'verify', 'publish'):
            result = runtime.run_stage(stage, item)
            assert result['status'] == 'success', (stage, result)
        query = CompanyProfileReadService(root).query(('600082.SH',))
        export = CompanyProfileReadService(root).export(('600082.SH',), export_directory=destination / 'export')
        reread = CompanyProfileReadService(root).query(('600082.SH',))
        assert query['profiles'] == export['profiles'] == reread['profiles']
        safe = CompanyProfileWorkpaper.model_validate(query['profiles'][0]['workpaper'])
        assert len(safe.quality_review.findings) == 14
        assert not any(f.resolution == 'corrected_pending_trust' for f in safe.quality_review.findings)
        assert safe.quality_review.status == 'incomplete'
        targets = provider.calls[1]['payload']['repair_targets']
        assert len(targets) == 12
        target = next(t for t in targets if t['target_fact_id'] == 'sc-01')
        assert len(target['additional_findings']) == 1
        removed = {f.affected_fact_id for f in safe.quality_review.findings if f.original_fact}
        originals = {f.fact_id: f.model_dump(mode='json') for e in (*seed.elements.model_dump().keys(),) for f in getattr(seed.elements, e).facts}
        if seed.supply_chain:
            originals.update({f.fact_id: f.model_dump(mode='json') for e in seed.supply_chain.model_dump().keys() for f in getattr(seed.supply_chain, e).facts})
        retained = [f for e in safe.elements.model_dump().keys() for f in getattr(safe.elements, e).facts]
        if safe.supply_chain:
            retained += [f for e in safe.supply_chain.model_dump().keys() for f in getattr(safe.supply_chain, e).facts]
        assert {f.fact_id for f in retained} == originals.keys() - removed
        assert all(f.model_dump(mode='json') == originals[f.fact_id] for f in retained)
        dump(destination / 'query.json', query)
        dump(destination / 'check.json', {
            'origin': 'controlled_saved_response_replay', 'new_model_requests': 0,
            'original_draft_hash_reconstructed': True, 'original_review_receipt_hash': initial['response_hash'],
            'source_and_original_results_not_modified': True, 'policy_version': safe.quality_review.policy_version,
            'findings_retained': 14, 'merged_repair_targets': 12, 'withheld_fact_ids': sorted(removed),
            'retained_facts': len(retained), 'unchanged_fact_objects': True, 'three_reads_equal': True,
            'no_repair_candidates_or_final_verification': True, 'original_real_run_still_failed': True,
        })
    finally:
        size = sum(p.stat().st_size for p in root.rglob('*') if p.is_file())
        shutil.rmtree(root)
        cleanup_path = DEST / 'local-check-cleanup.json'
        previous = read(cleanup_path) if cleanup_path.exists() else {}
        runs = previous.get('removed_runs', [previous] if previous else [])
        runs.append({'removed_root': str(root), 'removed_bytes': size, 'recoverable': 'rebuild from retained original responses and PDF'})
        dump(cleanup_path, {'removed_runs': runs, 'removed_bytes': sum(r['removed_bytes'] for r in runs)})
    print(json.dumps({'huangshan': check, 'haitai': read(destination / 'check.json')}, ensure_ascii=False))
