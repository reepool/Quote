def prepare(directory):
    started = time.monotonic()
    assert not (directory / '601318.json').exists()
    manifest = json.loads((CHANGE / 'workpaper-six-issue-regression-20261119/source-manifest.json').read_text())
    original = next(i for i in manifest['companies'] if i['instrument_id'] == '601318.SH')
    from research.company_profile.workpaper_sections import index_sections
    from research.business_profile_documents import classify_business_profile_document
    pdf = Path(original['asset']['archive_path'])
    assert sha(pdf) == original['asset']['content_hash']
    cached = pdf.parent / 'derived/business_profile_pdfium_recovery.v4' / (pdf.stem + '_57e1dd4f16ef40ff.json.gz')
    pages = json.loads(gzip.decompress(cached.read_bytes()))['pages']
    assert len(pages) == original['pdf_pages']
    binding_path = ROOT / original['source_binding']
    assert sha(binding_path) == original['source_binding_sha256']
    title = json.loads(binding_path.read_text())['announcement']['title']
    classified = classify_business_profile_document(title, adjunct_type='PDF')
    assert classified.selected and classified.is_full_report
    index = index_sections(pages)
    prepared = dict(asset=original['asset'], name=original['name'], exchange=original['exchange'], pages=pages,
                    index=[s.model_dump(mode='json') for s in index],
                    preparation_seconds=round(time.monotonic()-started, 3),
                    source_preparation=dict(reused_shared_cache=str(cached), cache_sha256=sha(cached)))
    dump(directory / '601318.json', prepared)
    item = {k: original[k] for k in ['instrument_id', 'name', 'exchange', 'asset', 'pdf_pages', 'source_binding', 'source_binding_sha256']}
    item.update(prepared_sha256=sha(directory / '601318.json'), heading_count=len(index), official_title=title)
    dump(DEST / 'source-manifest.json', dict(task='6.DW', companies=[item], old_semantic_inputs=False))
    check = json.loads((CHANGE / 'workpaper-section-isolation-20261119/source-check.json').read_text())
    for path, digest in check['source_code_sha256'].items():
        assert sha(ROOT / path) == digest
    prior_cleanup = json.loads((CHANGE / 'workpaper-section-isolation-20261119/cleanup.json').read_text())
    assert not Path(prior_cleanup['removed_root']).exists()
    assert not Path(prior_cleanup['interrupted_api_controlled_test_root']).exists()
    dump(DEST / 'readiness.json', dict(task='6.DW', new_model_requests=0, previous_owned_temp_absent=True,
        source_sha256=sha(pdf), prepared_sha256=sha(directory / '601318.json'), heading_count=len(index),
        reused_verification='993c7447:131 passed, no new semantic score', source_only=True,
        code_sha256=code_hashes(), preparation_seconds=prepared['preparation_seconds']))
    print('PREPARED_ONE_FRESH_SOURCE', len(index), flush=True)
