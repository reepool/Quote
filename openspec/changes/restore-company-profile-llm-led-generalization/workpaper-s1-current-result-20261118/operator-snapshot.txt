def network_preflight():
    import httpx
    import socket
    from urllib.parse import urlsplit
    config = config_for_run()
    checks = []
    for name in (EXTRACT, REVIEW):
        host = urlsplit(config.profiles[name].base_url).hostname
        assert host in ('api.deepseek.com', 'scorpio.reepool.com')
        begun = time.monotonic()
        addresses = socket.getaddrinfo(host, 443, type=socket.SOCK_STREAM)
        # Public TLS GET, no API keys, account data, prompts or paid model call.
        with httpx.Client(timeout=20, verify=True) as client:
            response = client.get('https://'+host+'/', follow_redirects=False)
            stream = response.extensions.get('network_stream')
            tls = stream.get_extra_info('ssl_object') if stream else None
            checks.append({'host': host, 'dns_addresses': len(addresses),
                'status_code': response.status_code, 'tls': tls.version() if tls else 'verified_by_httpx',
                'seconds': round(time.monotonic()-begun, 3)})
    dump(DEST/'network-preflight.json', {'task': '6.DL',
        'namespace': os.readlink('/proc/self/ns/net'), 'permission_mode': 'approved_network_exec',
        'checks': checks, 'model_requests': 0})
    print('NETWORK_PREFLIGHT_OK', flush=True)
