def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def read(path):
    return json.loads(Path(path).read_text())


def dump(path,value):
    Path(path).write_text(json.dumps(value,ensure_ascii=False,indent=2,default=str)+'\n')

async def verify_delivery():
    import httpx
    from fastapi import FastAPI
    from api.company_profile_routes import router
    os.environ['QUOTE_COMPANY_PROFILE_RESEARCH_ROOT']=str(OUT)
    app=FastAPI(); app.include_router(router,prefix='/api/v1')
    reader=CompanyProfileReadService(OUT)
    exported=reader.export(tuple(CHECKED),export_directory=DEST/'delivery')
    queried=reader.query(tuple(CHECKED))
    reloaded=CompanyProfileReadService(OUT).query(tuple(CHECKED))
    assert queried['profiles']==exported['profiles']==reloaded['profiles']
    summaries=[]
    async with httpx.AsyncClient(transport=httpx.ASGITransport(app=app),base_url='http://research-local') as client:
        for row in rows:
            instrument=row['instrument_id']
            response=await client.get('/api/v1/research/company/'+instrument+'/annual-profile')
            assert response.status_code==200 and response.json()==reader.research_profile(instrument)
            value=response.json()
            visible=[f for f in value['facts'] if f['fact'] is not None]
            assert all(f['fact'] is None for f in value['facts'] if f['state'] in ('source_only','withheld','duplicate'))
            if instrument=='600000.SH':
                assert value['readings'][0]['review_status']=='failed'
                assert any(f['state']=='pending' for f in value['facts'])
                assert all(next(f for f in value['facts'] if f['fact_id']==fid)['state']=='source_only' for fid in ('rm-06','rm-07'))
            summaries.append({'instrument_id':instrument,'api_status':response.status_code,
                'visible_records':len(visible),'metrics':sum(len(f['fact']['metrics']) for f in visible),
                'delivery_counts':dict(Counter(f['state'] for f in value['facts'])),
                'information_gaps':value['information_gaps'],'suggested_acquisitions':value['suggested_acquisitions'],
                'model_review_status':value['readings'][0]['review_status']})
    return summaries
