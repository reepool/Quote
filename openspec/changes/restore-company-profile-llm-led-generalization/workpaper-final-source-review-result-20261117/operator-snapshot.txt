def bind_controls(paper,row):
    from research.company_profile.workpaper import workpaper_fact_hash
    from research.company_profile.workpaper_delivery import bind_workpaper_use_review
    prior=next(p for p in read(CZ/'research-delivery.json')['profiles'] if p['instrument_id']==row['instrument_id'])
    controls={d['fact_id']:d for d in prior['use_review']['decisions']}
    if row['instrument_id']=='600104.SH':
        decision=read(CZ/'acceptance.json')['new_independent_decision'];controls[decision['fact_id']]=decision
    decisions=[]
    for fid,fact in facts(paper).items():
        old=controls.get(fid)
        if old is None:continue
        digest=workpaper_fact_hash(fact)
        if digest==old['fact_sha256']:
            decisions.append(old)
        elif old['disposition'] in {'withheld','source_only','duplicate'}:
            decisions.append({'fact_id':fid,'fact_sha256':digest,'disposition':'source_only',
                'evidence_ids':list(fact.evidence_ids),'reason':'本次修改了原受限对象，等待独立核原文；旧限制不由模型accepted自动解除。'})
    return bind_workpaper_use_review(paper,{'report':paper.report.model_dump(mode='json'),
        'reviewer':'Codex继承既有独立核源及限制；非本轮全量真值核验',
        'reference':str(DEST/'use-controls.json'),'reference_sha256':sha(DEST/'use-controls.json'),
        'decisions':decisions})

def group():
    frozen=read(DEST/'authorization.json');assert current_codes()==frozen['code_sha256']
    assert not (DEST/'group-clock.json').exists(),'this group is single-use'
    started=time.monotonic();deadline=started+7200
    dump(DEST/'group-clock.json',{'supervision_started':started,'absolute_deadline':deadline})
    pending=list(frozen['order']);processes={};logs={};results={};stop=None;signalled=set()
    supervision=Path(tempfile.mkdtemp(prefix='workpaper-dc-supervision-'))
    try:
        while pending or len(results)<len(processes):
            for instrument,process in processes.items():
                if instrument in results:continue
                code=instrument[:6];timing=DEST/code/'entry-timing.json'
                if process.poll() is None and timing.exists():
                    end=read(timing)['absolute_deadline_monotonic']
                    if time.monotonic()>=end and instrument not in signalled:
                        process.send_signal(signal.SIGUSR1);signalled.add(instrument)
                    if time.monotonic()>=end+30:process.terminate()
                if process.poll() is not None:
                    logs[instrument].close();path=DEST/code/'summary.json'
                    results[instrument]=read(path) if path.exists() else {'instrument_id':instrument,'exception':'operator_exit_'+str(process.returncode),'readable':False,'failure_signature':'operator_exit_'+str(process.returncode)}
                    print('COMPANY_DONE '+json.dumps({k:results[instrument].get(k) for k in ('instrument_id','readable','review_status','fact_count','physical_requests','charged_tokens','business_seconds','exception')},ensure_ascii=False),flush=True)
            ordered=[results.get(i) for i in frozen['order']]
            if time.monotonic()>=deadline:stop=stop or 'group_deadline'
            if (DEST/'independent-stop.json').exists():stop=stop or 'unisolatable_core_contamination'
            for left,right in zip(ordered,ordered[1:]):
                if left and right and left.get('failure_signature') and left['failure_signature']==right.get('failure_signature'):
                    stop=stop or 'consecutive_same_core_failure:'+left['failure_signature']
            totals={'physical_requests':0,'charged_tokens':0}
            for instrument in processes:
                path=DEST/instrument[:6]/'budget.json'
                if path.exists():
                    try:snapshot=read(path)
                    except json.JSONDecodeError:continue
                    for key in totals:totals[key]+=snapshot[key]
            if totals['physical_requests']>=72 or totals['charged_tokens']>=6000000:stop=stop or 'group_resource_limit'
            if stop in {'group_deadline','unisolatable_core_contamination','group_resource_limit'}:
                for instrument,process in processes.items():
                    if process.poll() is None and instrument not in signalled:
                        process.send_signal(signal.SIGUSR1);signalled.add(instrument)
            while pending and stop is None and sum(p.poll() is None for p in processes.values())<4:
                instrument=pending.pop(0);code=instrument[:6]
                logs[instrument]=(supervision/(code+'-run.log')).open('x')
                processes[instrument]=subprocess.Popen([sys.executable,__file__,'run','--symbol',code,'--group-deadline',str(deadline),'--supervision',str(supervision)],
                    stdout=logs[instrument],stderr=subprocess.STDOUT)
                print('DISPATCH '+instrument,flush=True)
            dump(DEST/'progress.json',{'executed':len(results),'dispatched':len(processes),'not_started':pending,
                'results':[r for r in ordered if r],'stop_reason':stop,'totals':totals,
                'supervision_seconds':round(time.monotonic()-started,3)})
            if stop and len(results)==len(processes):break
            if pending or len(results)<len(processes):time.sleep(1)
        dump(DEST/'group-execution.json',{'results':[results[i] for i in frozen['order'] if i in results],
            'not_started':pending,'stop_reason':stop,'source_code_unchanged':current_codes()==frozen['code_sha256'],
            'supervision_seconds':round(time.monotonic()-started,3)})
    finally:
        assert all(p.poll() is not None for p in processes.values()),'keep live workers and their owned roots'
        # Persist only a short tail for abnormal worker exits; normal receipts are authoritative.
        for instrument,result in results.items():
            if 'operator_exit_' in (result.get('exception') or ''):
                path=supervision/(instrument[:6]+'-run.log')
                (DEST/instrument[:6]).mkdir(parents=True,exist_ok=True)
                (DEST/instrument[:6]/'operator-failure.txt').write_text(path.read_text()[-12000:])
                ownership=supervision/(instrument[:6]+'-root.json')
                if ownership.exists():
                    abandoned=Path(read(ownership)['root'])
                    assert abandoned.parent==ROOT/'data/QuoteBak' and abandoned.name.startswith('workpaper-dc-'+instrument[:6]+'-')
                    if abandoned.exists():
                        for folder in ('captured-content','output','checkpoints'):
                            for saved in (abandoned/folder).rglob('*.json'):
