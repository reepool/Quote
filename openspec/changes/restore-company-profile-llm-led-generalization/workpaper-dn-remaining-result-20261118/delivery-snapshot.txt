def read(path):return json.loads(Path(path).read_text())
def sha(path):return hashlib.sha256(Path(path).read_bytes()).hexdigest()
def dump(path,value):Path(path).write_text(json.dumps(value,ensure_ascii=False,indent=2,default=str)+'\n')
def facts(paper):return {f.fact_id:f for e in workpaper_elements(paper).values() for f in e.facts}

assert not OUT.exists() and not CURRENT.exists() and not (D/'independent-audit.json').exists()
auth=read(D/'authorization.json');group=read(D/'group-execution.json');notes=read(D/'audit-notes.json')
assert group['executed']==4 and group['not_started']==[] and group['stop_reason'] is None
assert read(D/'group-clock.json')['deadline_monotonic']==read(PRIOR/'group-clock.json')['deadline_monotonic']
reference=read(ROOT/auth['source_first_reference']);refs={c['instrument_id']:c for c in reference['companies']}
contexts={};papers={};inputs={};audits=[]
for note in notes['companies']:
    instrument=note['instrument_id'];folder=D/instrument[:6]
    for name in ('query.json','runtime.json','budget.json','model-receipts.json','summary.json'):
        inputs[str((folder/name).relative_to(ROOT))]=sha(folder/name)
    record=CompanyProfileRuntimeRecordV3.model_validate_json((folder/'runtime.json').read_text());paper=record.workpaper
    current=facts(paper);accepted={f.revised_fact.fact_id for f in paper.quality_review.findings if f.resolution=='corrected_pending_trust'}
    assert accepted==set(note['accepted_revision_fact_ids'])
    assert all(f.revised_fact==current[f.revised_fact.fact_id] for f in paper.quality_review.findings if f.resolution=='corrected_pending_trust')
    assert len(note['basic_condition_fact_ids'])==len(refs[instrument]['core_conditions'])
    checked=set(accepted)
    for ids in note['basic_condition_fact_ids']:checked.update(ids)
    assert checked<=current.keys() and not (set(note.get('rejected_originals_absent',[])) & current.keys())
    decisions=[{'fact_id':fid,'fact_sha256':workpaper_fact_hash(current[fid]),'disposition':'source_checked',
        'evidence_ids':list(current[fid].evidence_ids),'reason':note['source_reason'],
        'limitation':note['limitations'].get(fid)} for fid in sorted(checked)]
    conditions=[{'condition_id':condition['condition_id'],'verdict':'detail_gap' if condition['condition_id']==note.get('detail_gap_condition') else 'supported',
        'fact_ids':ids} for condition,ids in zip(refs[instrument]['core_conditions'],note['basic_condition_fact_ids'])]
    audits.append({'instrument_id':instrument,'basic_conditions':conditions,'source_checked_fact_ids':sorted(checked),
        'accepted_repairs_independently_checked':sorted(accepted),'decisions':decisions,'remaining_scope':'not_whole_output_accuracy'})
    contexts[instrument]=record;papers[instrument]=paper
audit={'task':'6.DN remaining four','method':'Independent source-context comparison plus frozen same-PDF basic reference; not model agreement.',
    'notes_sha256':sha(D/'audit-notes.json'),'source_reference_sha256':auth['source_first_reference_sha256'],
    'basic_supported':19,'basic_detail_gaps':1,'basic_assessed':20,'accepted_repairs_independently_checked':13,
    'source_checked_objects':sum(len(a['source_checked_fact_ids']) for a in audits),'companies':audits,
    'raw_inputs_sha256_unchanged':inputs,'all_output_accuracy':None,'formal_S1_pass':False,'production_authorization':'not_authorized'}
assert sum(len(a['accepted_repairs_independently_checked']) for a in audits)==13
dump(D/'independent-audit.json',audit)
writer=CompanyProfileResearchWriter(OUT);rows=[]
for note,company in zip(notes['companies'],audits):
    instrument=company['instrument_id'];paper=papers[instrument]
    bound=bind_workpaper_use_review(paper,{'report':paper.report.model_dump(mode='json'),
        'reviewer':'Codex: actual accepted repairs and basic questions only; company detail gaps remain usable',
        'reference':str((D/'independent-audit.json').relative_to(ROOT)),
        'reference_sha256':sha(D/'independent-audit.json'),'decisions':company['decisions']})
    if note.get('program_gap_note'):
        product=bound.elements.products_services
        product=product.model_copy(update={'limitations':(*product.limitations,note['program_gap_note'])})
        bound=bound.model_copy(update={'elements':bound.elements.model_copy(update={'products_services':product})})
    assert facts(bound)==facts(paper) and bound.quality_review==paper.quality_review and bound.generation==paper.generation and bound.evidence==paper.evidence
    papers[instrument]=bound
    path=writer.persist(contexts[instrument].model_copy(update={'workpaper':bound}))
    rows.append({'instrument_id':instrument,'runtime_record':str(path),'runtime_sha256':sha(path),
        'actual_provider_cap_seconds':2450,'resource_binding_version':'unused_four_provider_2450'})

async def verify():
    import httpx
    from fastapi import FastAPI
    from api.company_profile_routes import router
    app=FastAPI();app.include_router(router,prefix='/api/v1')
    results=[];payloads=[];four_summary=[]
    with tempfile.TemporaryDirectory(prefix='workpaper-dn-final-exports-') as temp:
        for root,instruments in ((OUT,tuple(papers)),(CURRENT,tuple(sorted(selected)))):
            reader=CompanyProfileReadService(root);query=reader.query(instruments)
            exported=reader.export(instruments,export_directory=Path(temp)/root.name)
            fresh=CompanyProfileReadService(root).query(instruments)
            assert query['profiles']==exported['profiles']==fresh['profiles']
            os.environ['QUOTE_COMPANY_PROFILE_RESEARCH_ROOT']=str(root)
            async with httpx.AsyncClient(transport=httpx.ASGITransport(app=app),base_url='http://research-local') as client:
                for instrument in instruments:
                    response=await client.get('/api/v1/research/company/'+instrument+'/annual-profile')
                    assert response.status_code==200 and response.json()==reader.research_profile(instrument)
                    value=response.json();visible=[f for f in value['facts'] if f['fact'] is not None]
                    assert all(f['fact'] is None for f in value['facts'] if f['state'] in {'source_only','withheld','duplicate'})
                    summary={'instrument_id':instrument,'api_status':200,'visible_records':len(visible),
                        'metrics':sum(len(f['fact']['metrics']) for f in visible),'delivery_counts':dict(Counter(f['state'] for f in value['facts'])),
                        'review_status':value['readings'][0]['review_status'],'information_gaps':value['information_gaps']}
                    if instrument=='600082.SH':assert summary['review_status']=='incomplete'
                    if instrument=='600104.SH':assert any('零部件' in reason for gap in value['information_gaps'] for reason in gap['reasons'])
                    if instrument=='600097.SH':assert 'F-OD-03' not in {f['fact_id'] for f in value['facts']}
                    if root==OUT:
                        four_summary.append(summary)
                        profile=next(p for p in query['profiles'] if p['instrument_id']==instrument)
                        (D/instrument[:6]/'profile.md').write_text(render_workpaper(papers[instrument],delivery=profile['workpaper_delivery']))
                    else:
                        results.append(summary);payloads.append(value)
    dump(D/'current-research-profiles.json',{'schema_version':'company_profile_current_research_snapshot.v1',
        'research_only':True,'formal_S1_pass':False,'production_authorization':'not_authorized',
        'not_twelve_new_runs_or_uniform_resource_binding':True,'profiles':payloads})
    return four_summary,results
