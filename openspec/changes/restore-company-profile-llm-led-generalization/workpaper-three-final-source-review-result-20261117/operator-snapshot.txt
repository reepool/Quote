def bind_controls(paper,instrument):
    from research.company_profile.workpaper import workpaper_fact_hash
    from research.company_profile.workpaper_delivery import bind_workpaper_use_review
    prior=next(p for p in read(DC/'delivery-decisions.json')['profiles'] if p['instrument_id']==instrument)['use_review']
    current=facts(paper);decisions=[]
    for old in prior['decisions']:
        fid=old['fact_id']
        if fid not in current:continue
        f=current[fid];digest=workpaper_fact_hash(f)
        if digest==old['fact_sha256']:decisions.append(old)
        elif old['disposition'] in {'withheld','source_only','duplicate'}:
            decisions.append({'fact_id':fid,'fact_sha256':digest,'disposition':'source_only',
                'evidence_ids':list(f.evidence_ids),'reason':'原独立限制对象已改变，实际新核源前不由模型accepted解除。'})
    return bind_workpaper_use_review(paper,{'report':paper.report.model_dump(mode='json'),
        'reviewer':'Codex继承DC逐内容hash独立限制；非本次全量源核',
        'reference':str(DC/'delivery-decisions.json'),'reference_sha256':sha(DC/'delivery-decisions.json'),'decisions':decisions})

def group():
    frozen=read(DEST/'authorization.json');assert H['code_hashes']()==frozen['code_sha256']
    assert not (DEST/'group-clock.json').exists(),'approved group already started'
    start=time.monotonic();deadline=start+1200
    dump(DEST/'group-clock.json',{'started':start,'deadline':deadline})
    supervision=Path(tempfile.mkdtemp(prefix='workpaper-de-supervision-'));processes={};logs={};signalled=set()
    try:
        for instrument in frozen['order']:
            code=instrument[:6];logs[code]=(supervision/(code+'.log')).open('x')
            processes[code]=subprocess.Popen([sys.executable,__file__,'run','--symbol',code,'--deadline',str(deadline),'--supervision',str(supervision)],stdout=logs[code],stderr=subprocess.STDOUT)
            print('DISPATCH '+code,flush=True)
        while any(p.poll() is None for p in processes.values()):
            for code,p in processes.items():
                timing=DEST/code/'entry-timing.json'
                end=min(deadline,read(timing)['deadline']) if timing.exists() else deadline
                if p.poll() is None and time.monotonic()>=end and code not in signalled:
                    p.send_signal(signal.SIGUSR1);signalled.add(code)
                if p.poll() is None and time.monotonic()>=end+30:p.terminate()
            time.sleep(1)
        results=[]
        for code,p in processes.items():
            logs[code].close();path=DEST/code/'summary.json'
            if path.exists():results.append(read(path))
            else:
                dump(DEST/code/'operator-failure.json',{'returncode':p.returncode,'log_tail':(supervision/(code+'.log')).read_text()[-12000:]})
                results.append({'instrument_id':code+'.SH','readable':False,'exception':'operator_exit_'+str(p.returncode)})
            print('COMPANY_DONE '+json.dumps(results[-1],ensure_ascii=False),flush=True)
        assert sum(r.get('physical_requests',0) for r in results)<=6 and sum(r.get('ledger',0) for r in results)<=900000
        dump(DEST/'group-execution.json',{'results':results,'supervision_seconds':round(time.monotonic()-start,3),
            'source_code_unchanged':H['code_hashes']()==frozen['code_sha256'],
            'shared_config_unchanged':sha(ROOT/'config/13_llm.json')==frozen['shared_config_sha256']})
    finally:
        assert all(p.poll() is not None for p in processes.values())
        for code in processes:
            ownership=supervision/(code+'-root.json')
            if ownership.exists():
                abandoned=Path(read(ownership)['root'])
                if abandoned.exists():
                    assert abandoned.parent==ROOT/'data/QuoteBak' and abandoned.name.startswith('workpaper-de-'+code+'-')
                    for folder in ('captured-content','output','checkpoints'):
                        for src in (abandoned/folder).rglob('*.json'):
                            dst=DEST/code/'interrupted-evidence'/src.relative_to(abandoned);dst.parent.mkdir(parents=True,exist_ok=True);shutil.copy2(src,dst)
                    size=sum(x.stat().st_size for x in abandoned.rglob('*') if x.is_file());shutil.rmtree(abandoned)
                    dump(DEST/code/'interrupted-cleanup.json',{'removed_bytes':size})
        size=sum(p.stat().st_size for p in supervision.rglob('*') if p.is_file());shutil.rmtree(supervision)
        dump(DEST/'supervision-cleanup.json',{'removed_bytes':size})
