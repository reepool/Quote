"""Read-only checks of the saved EA artifacts/API under the new local decoder."""
import hashlib
import json
from pathlib import Path
import sys

ROOT = Path('/home/python/Quote')
sys.path.insert(0, str(ROOT))
from research.company_profile.reads import CompanyProfileReadService

EA = ROOT / 'openspec/changes/restore-company-profile-llm-led-generalization/workpaper-s2-confirmation-result-20261119'
index = json.loads((EA / 'index.json').read_text())
for name, metadata in index['files'].items():
    assert hashlib.sha256((EA / name).read_bytes()).hexdigest() == metadata['sha256'], name
for name, expected in index['research_runtime_files'].items():
    assert hashlib.sha256((ROOT / name).read_bytes()).hexdigest() == expected, name
saved = json.loads((EA / 'delivery/annual-api.json').read_text())
reader = CompanyProfileReadService(ROOT / 'data/research/company_profile_s2_confirmation_20261119')
assert reader.research_profile_catalog() == saved['catalog']
for detail in saved['profiles']:
    assert reader.research_profile(detail['instrument_id']) == detail
print(json.dumps(dict(canonical_files_unchanged=len(index['files']), writer_files_unchanged=len(index['research_runtime_files']),
    old_API_profiles_equal=len(saved['profiles']), old_catalog_equal=True, new_model_requests=0), ensure_ascii=False))
