def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def source(code):
    item = next(row for row in json.loads((OLD / 'source-manifest.json').read_text())['companies'] if row['instrument_id'].startswith(code))
    asset = item['asset']
    pdf = Path(asset['archive_path'])
    assert sha(pdf) == asset['content_hash']
    cache = pdf.parent / 'derived/business_profile_pdfium_recovery.v4' / (pdf.stem + '_57e1dd4f16ef40ff.json.gz')
    pages = json.loads(gzip.decompress(cache.read_bytes()))['pages']
    report = ReportIdentity(instrument_id=asset['instrument_id'], report_id=asset['filing_id'], document_version=pdf.stem,
                            report_period=asset['report_period'], published_at=asset['published_at'])
    return pdf, pages, report


def packet_summary(packet):
    return {'section_titles': [s.section_path for s in packet.reading_scope.sections],
            'physical_pages': sorted({s.page for e in packet.evidence for s in e.source_spans}),
            'source_characters': sum(len(s.quote) for e in packet.evidence for s in e.source_spans),
            'transcription_version': packet.transcription_version, 'limitations': packet.limitations,
            'model_text_sha256': hashlib.sha256(packet.model_text().encode()).hexdigest()}

from pathlib import Path

               for name in ['evidence.py', 'pytest-final', 'pytest-closed', 'pytest-native-gap'])
verification = json.loads((DEST / 'verification.json').read_text())
for path in TEMP.glob('*.xml'):
    suite = ET.parse(path).getroot().find('testsuite')
    verification['tests'][path.name] = {key: suite.get(key) for key in ['tests', 'failures', 'errors', 'skipped', 'time']}
    failures = [{'test': case.get('name'), 'message': case.find('failure').get('message')}
                for case in suite.findall('testcase') if case.find('failure') is not None]
    if failures:
        verification['tests'][path.name]['failure_details'] = failures
assert verification['tests']['closed-tests.xml']['failures'] == '0'
assert verification['tests']['native-gap-tests.xml']['failures'] == '0'
verification['baseline_commit'] = 'ec855b46'
(DEST / 'verification.json').write_text(json.dumps(verification, ensure_ascii=False, indent=2) + '\n')
(DEST / 'evidence-snapshot.txt').write_text((TEMP / 'evidence.py').read_text() + '\n\n# Closing operation\n' + Path(__file__).read_text())
for name, expected in json.loads((OLD / 'readback-check.json').read_text())['raw_runtime_sha256_unchanged'].items():
    assert hashlib.sha256((ROOT / name).read_bytes()).hexdigest() == expected
files = [p for p in TEMP.rglob('*') if p.is_file()]
record = {'removed_root': str(TEMP), 'files': len(files), 'file_bytes': sum(p.stat().st_size for p in files),
          'previous_finally_cleanup_bytes': 'not_aggregated_do_not_double_count',
          'retained': ['official PDF/shared valid cache', 'original S2 runtime/query/failures/API/attempts',
                       'current verification/result and exact evidence script snapshot', 'baseline and parallel changes'],
          'reproducible_derivatives_only': True}
shutil.rmtree(TEMP)
(DEST / 'cleanup.json').write_text(json.dumps(record, ensure_ascii=False, indent=2) + '\n')
print(json.dumps(record, ensure_ascii=False))
