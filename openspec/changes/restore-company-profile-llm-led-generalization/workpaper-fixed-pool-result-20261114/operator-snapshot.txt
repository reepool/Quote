def config_for_run():
    from utils.llm.models import LlmConfig
    from utils.llm.config import load_project_environment
    load_project_environment(ROOT)
    original=LlmConfig.from_mapping(json.loads((ROOT/'config/13_llm.json').read_text())['llm'])
    profiles={}

        profile=original.profiles[name]
        assert original.enabled and profile.enabled and os.environ.get(profile.api_key_env)
        assert name not in original.routes
        profiles[name]=replace(profile,timeout_seconds=620,attempt_timeout_seconds=600,max_retries=1,max_schema_repair_attempts=1)
    assert original.resource_for_profile(profiles[EXTRACT]).name=='deepseek'

    assert urlsplit(profiles[REVIEW].base_url).hostname=='scorpio.reepool.com'
    assert profiles[REVIEW].model=='grok-4.7' and profiles[REVIEW].reasoning_effort=='high'
    return replace(original,profiles=profiles,routes={},pools={})

def code_hashes():
    paths=sorted((ROOT/'research/company_profile').glob('*.py'))+sorted((ROOT/'utils/llm').glob('*.py'))
    paths += [ROOT/'research/storage.py',ROOT/'research/business_profile_pdf_artifacts.py',ROOT/'research/business_profile_production_operations.py']
    return {str(p.relative_to(ROOT)):sha(p) for p in paths}

def dump(path, value):
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2, default=str) + '\n')

def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def source_items():
    return json.loads(MANIFEST.read_text())['companies']

def prepare_one(item, directory):
    from research.business_profile_pdf_artifacts import ensure_archived_pdf_page_artifact
    from research.company_profile.workpaper_sections import index_sections
    started=time.monotonic(); asset=item['asset']; symbol=item['instrument_id'][:6]
    assert sha(Path(asset['archive_path'])) == asset['content_hash']
    parsed=ensure_archived_pdf_page_artifact(asset)
    artifact=parsed['artifact']
    pages=[{'page':p.page_number,'text':p.text} for p in artifact.pages]
    assert len(pages)==item['pdf_pages']
    index=index_sections(pages)
    prepared={'asset':asset,'name':item['name'],'pages':pages,
              'index':[s.model_dump(mode='json') for s in index],
              'preparation_seconds':round(time.monotonic()-started,3),
              'source_preparation':{k:parsed.get(k) for k in ('artifact_path','artifact_hash','status','cache_status','timings')}}
    dump(directory/(symbol+'.json'),prepared)
    candidates=[{'id':s.section_id,'title':' / '.join(s.section_path),'pages':[s.start_page,s.end_page], 'characters':s.characters}
                for s in index if s.start_page<=80 and re.search('从事的业务|主要业务|主营业务|经营模式|经营情况讨论|经营情况分析|公司总体经营|收入和成本分析|分部信息|主要会计数据|主要财务指标|利润表分析',s.section_path[-1])]
    print(json.dumps({'prepared':symbol,'pages':len(pages),'source_seconds':prepared['preparation_seconds'],'sections':candidates},ensure_ascii=False),flush=True)
    return symbol

def prepare():
    assert os.environ.get('TMPDIR')==str(ROOT/'data/QuoteBak')
    directory=Path(tempfile.mkdtemp(prefix='workpaper-bu-prepared-'))
    print('PREPARED_ROOT='+str(directory),flush=True)
    with ThreadPoolExecutor(max_workers=4) as pool:
        list(pool.map(lambda item:prepare_one(item,directory),source_items()))
    print('PREPARE_DONE='+str(directory),flush=True)

def inspect(directory, symbol, pages=None, sections=None):
    prepared=json.loads((directory/(symbol+'.json')).read_text())
    if sections:
        for entry in prepared['index']:
            if re.search(sections,entry['section_path'][-1]):
                print(json.dumps({k:entry[k] for k in ('section_id','section_path','start_page','end_page','characters')},ensure_ascii=False))
        return
    from research.company_profile.workpaper_sections import extract_selected_layout
    page_numbers=[int(v) for v in pages.split(',')]
    for page in extract_selected_layout(Path(prepared['asset']['archive_path']).read_bytes(), page_numbers):
        print('\nPHYSICAL PAGE',symbol,page['page'])
        print(page['text'])

def run_one(directory,symbol,group_deadline):
    from research.announcements import AnnouncementAttachment, AnnouncementRecord
    from research.business_profile_production_operations import BusinessProfileAnnouncementFrontierRepository
    from research.company_profile.models import ReportIdentity
    from research.company_profile.operations import CompanyProfileTaskService
    from research.company_profile.reads import CompanyProfileReadService

    from research.company_profile.workpaper_llm import workpaper_processing_identity
    from research.storage import ResearchStorageManager
    from utils.config_manager import ResearchConfig,ResearchStorageConfig
    from utils.llm.client import LlmClient
    from utils.llm.budget import LlmRunBudget, LogicalRecoveryPolicy, DEVELOPMENT_UNPRICED_BILLING
    preparation=time.monotonic(); frozen=json.loads((DEST/'authorization.json').read_text())
    assert code_hashes()==frozen['code_sha256'] and sha(ROOT/'config/13_llm.json')==frozen['shared_config_sha256']
    assert sha(Path(__file__))==frozen['operator_sha256']
    config=config_for_run(); source=json.loads((directory/(symbol+'.json')).read_text()); asset=source['asset']
    assert asset['instrument_id'] in frozen['order'] and sha(Path(asset['archive_path']))==asset['content_hash']
    item=next(i for i in source_items() if i['instrument_id']==asset['instrument_id'])
    announcement=json.loads((ROOT/item['source_binding']).read_text())['announcement']
    report=ReportIdentity(instrument_id=asset['instrument_id'],report_id=asset['filing_id'],document_version=asset['content_hash'],report_period=asset['report_period'],published_at=asset['published_at'])
    run_id='v9-fixed-pool-'+symbol+'-20261114'; destination=CHANGE/'workpaper-runs'/run_id
    assert not destination.exists(); destination.mkdir(parents=True)
    root=Path(tempfile.mkdtemp(prefix='workpaper-bu-'+symbol+'-')); client=None; budget=None; service=None
    summary={'instrument_id':asset['instrument_id'],'name':source['name'],'run_id':run_id,'three_reads_equal':False,'exception':None}
    started=None
    try:
        db=str(root/'research.db')
        storage=ResearchStorageManager(ResearchConfig(storage=ResearchStorageConfig(db_path=db,financials_db_path=db,valuation_db_path=db,interests_db_path=db,quotes_db_path=str(root/'quotes.db'),attach_quotes_db=False)))
        storage.initialize()
        BusinessProfileAnnouncementFrontierRepository(storage).upsert_record(instrument={'instrument_id':asset['instrument_id'],'symbol':symbol,'exchange':'SSE'},
            record=AnnouncementRecord(source='cninfo',source_announcement_id=asset['filing_id'],announcement_key='cninfo:'+asset['filing_id'],title=announcement['title'],published_at=asset['published_at'],exchange='SSE',symbols=(symbol,),attachments=(AnnouncementAttachment(source_url=announcement['source_url'],resolved_url=announcement['source_url'],file_extension='PDF'),)))
        identity=workpaper_processing_identity(real=True,review=True,review_profile=REVIEW)
        identity.update(review_max_output_tokens=16384,repair_profile=EXTRACT)
        # All setup above precedes the business deadline. No model is callable yet.
        service=CompanyProfileTaskService(storage=storage,output_root=root/'output',checkpoint_root=root/'checkpoints',page_source=lambda _: {'report':report,'pages':source['pages'],'source_pdf_path':asset['archive_path']},processing_identity=identity,token_budget=500000)
        binding={**frozen,'run_id':run_id,'asset':asset,'physical_pages':len(source['pages']),'identity':identity,'source_preparation':source['source_preparation'],'source_only_metadata_from':item['source_binding']}
        binding.pop('prepared_root',None); binding.pop('order',None)
        dump(destination/'binding.json',binding)
        marker=ROOT/'data/QuoteBak/s1-attempts'/('workpaper-bu-'+symbol+'-20261114.json')
        marker.parent.mkdir(parents=True,exist_ok=True)
        with marker.open('x') as handle: json.dump({'run_id':run_id,'approval_reference':frozen['approval_reference'],'binding_sha256':sha(destination/'binding.json')},handle,ensure_ascii=False)
        assert time.monotonic()<group_deadline
        started=time.monotonic(); summary['preparation_seconds']=round(started-preparation+source['preparation_seconds'],3)
        budget=LlmRunBudget(deadline=min(started+1800,group_deadline),prices={},max_tokens=500000,max_requests=10,max_cost_cny=Decimal(10),ledger_path=destination/'budget.json',billing_mode=DEVELOPMENT_UNPRICED_BILLING,recovery_policy=LogicalRecoveryPolicy(3,1),run_authorization='company_profile_workpaper_development.2026-11-11')
        client=LlmClient(config,physical_budget=budget)
        provider=CommonGatewaySemanticProvider(client=client,profile=EXTRACT,max_output_tokens=16384,timeout_seconds=620,execution_authorization={'status':'authorized','document_version':asset['content_hash'],'approval_reference':frozen['approval_reference']})
        service.provider=provider; service.runtime.provider=provider; service.execution_budget=budget; service.runtime.execution_budget=budget
        print('BUSINESS_STARTED '+symbol,flush=True)
        try:
            executed=asyncio.run(service.execute('run',knowledge_cutoff='2026-11-14',instrument_ids=(asset['instrument_id'],),max_items=1,max_elapsed_seconds=1800))
            dump(destination/'execution.json',executed)
        except Exception as exc:
            summary['exception']=type(exc).__name__+': '+str(exc)[:3000]

        query=asyncio.run(service.execute('query',instrument_ids=(asset['instrument_id'],)))

        dump(destination/'query.json',query); dump(destination/'model-receipts.json',provider.workpaper_receipts)

        else: summary.update(readable=False,facts=0,review_status=None)
        missing=[k for k in ('business_structure','products_services','revenue_model') if not (summary.get('element_counts') or {}).get(k)]
        summary['core_missing_elements']=missing
        failure=summary.get('review_error') if summary.get('review_status') in ('failed','incomplete') else summary.get('exception')
        if missing and not failure:
            errors=summary['contract_errors'] or summary['failed_calls']
            failure=json.dumps(errors[-1],ensure_ascii=False,sort_keys=True) if errors else 'missing_core_elements:'+','.join(missing)
        summary['failure_signature']=re.sub(r'[0-9a-f]{16,}|\d+', '#', failure or '')[:800] or None
        snapshot=budget.snapshot(); dump(destination/'budget.json',snapshot)
        summary.update(physical_requests=snapshot['physical_requests'],charged_tokens=snapshot['charged_tokens'],cost=snapshot['charged_cost_cny'],stopped_reason=snapshot['stopped_reason'])
        counts={}
        for a in snapshot['attempts']:
            key=a.get('logical_request_id') or a['request_id']; counts[key]=counts.get(key,0)+1
        summary['recoveries']=sum(n-1 for n in counts.values())
        assert summary['three_reads_equal'] and summary['recoveries']<=3 and all(n<=2 for n in counts.values())
        assert code_hashes()==frozen['code_sha256'] and sha(ROOT/'config/13_llm.json')==frozen['shared_config_sha256']
    except Exception as exc:
        summary['exception']=type(exc).__name__+': '+str(exc)[:3000]
        if budget:
            snapshot=budget.snapshot(); dump(destination/'budget.json',snapshot)
            summary.update(physical_requests=snapshot['physical_requests'],charged_tokens=snapshot['charged_tokens'],cost=snapshot['charged_cost_cny'])
    finally:
        if client:
            try: asyncio.run(client.close())
            except Exception as exc: summary['close_error']=type(exc).__name__
        summary['supervision_seconds']=round(time.monotonic()-started,3) if started else None
        summary['cleanup_bytes']=sum(p.stat().st_size for p in root.rglob('*') if p.is_file())
        assert root.parent==ROOT/'data/QuoteBak' and root.name.startswith('workpaper-bu-'+symbol+'-')
        shutil.rmtree(root)
        dump(destination/'summary.json',summary)
        print('RESULT '+json.dumps(summary,ensure_ascii=False),flush=True)

def group(directory):
    deadline=time.monotonic()+7200; group_start=time.monotonic(); results=[]; stop=None
    items=source_items()
    for first in range(0,12,4):
        if time.monotonic()>=deadline: stop='group_deadline'; break
        def launch(item):
            symbol=item['instrument_id'][:6]; log=directory/(symbol+'-run.log')
            with log.open('w') as output:
                process=subprocess.run([sys.executable,str(Path(__file__)),'run','--prepared',str(directory),'--symbol',symbol,'--group-deadline',str(deadline)],stdout=output,stderr=subprocess.STDOUT,timeout=min(2100,max(1,deadline-time.monotonic())))
            result_path=CHANGE/'workpaper-runs'/('v9-fixed-pool-'+symbol+'-20261114')/'summary.json'
            result=json.loads(result_path.read_text()) if result_path.exists() else {'instrument_id':item['instrument_id'],'exception':'operator_exit_'+str(process.returncode),'readable':False}
            print('COMPANY_DONE '+json.dumps(result,ensure_ascii=False),flush=True)
            return result
        with ThreadPoolExecutor(max_workers=4) as pool: batch=list(pool.map(launch,items[first:first+4]))
        results.extend(batch)
        for left,right in zip(results,results[1:]):
            # Match a concrete application failure or review failure, not core_complete=false
            # (all draft workpapers intentionally retain false trusted flags).
            a=left.get('failure_signature') or left.get('exception')
            b=right.get('failure_signature') or right.get('exception')
            if a and b and a==b: stop='consecutive_same_core_failure: '+a; break
        dump(DEST/'progress.json',{'executed':len(results),'results':results,'stop_reason':stop,'business_wall_seconds':round(time.monotonic()-group_start,3)})
        if stop: break
    dump(DEST/'group-execution.json',{'executed':len(results),'not_started':[i['instrument_id'] for i in items[len(results):]],'results':results,'stop_reason':stop,'business_wall_seconds':round(time.monotonic()-group_start,3),'source_code_unchanged':code_hashes()==json.loads((DEST/'authorization.json').read_text())['code_sha256']})
    print('GROUP_DONE '+str(len(results))+' '+str(stop),flush=True)

def main():
    parser=argparse.ArgumentParser(); parser.add_argument('mode',choices=['prepare','inspect','freeze','run','group'])
    parser.add_argument('--prepared',type=Path); parser.add_argument('--symbol'); parser.add_argument('--pages'); parser.add_argument('--sections')
    parser.add_argument('--group-deadline',type=float)
    args=parser.parse_args()
    if args.mode=='prepare': prepare()
    elif args.mode=='inspect': inspect(args.prepared,args.symbol,args.pages,args.sections)
    elif args.mode=='freeze': freeze(args.prepared)
    elif args.mode=='run': run_one(args.prepared,args.symbol,args.group_deadline)
    else: group(args.prepared)

if __name__=='__main__':main()
