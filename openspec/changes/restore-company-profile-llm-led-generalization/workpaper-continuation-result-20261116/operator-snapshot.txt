def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def digest(value):
    return hashlib.sha256(json.dumps(value, ensure_ascii=False, sort_keys=True).encode()).hexdigest()


def dump(path, value):
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2, default=str)+'\n')


def limits(code):

            'requests': 8 if code=='600098' else 4,
            'recoveries': 3 if code=='600098' else 2}

def group(directory):
    started=time.monotonic();deadline=started+2400
    processes={};handles={}
    try:
        for code in ORDER:
            log=(directory/(code+'-run.log')).open('x');handles[code]=log
            processes[code]=subprocess.Popen([sys.executable,str(Path(__file__)),'run',str(directory),code,str(deadline)],
                cwd=ROOT,stdout=log,stderr=subprocess.STDOUT,env=os.environ.copy())
        dump(DEST/'group-clock.json',{'supervision_started_monotonic':started,'group_deadline_monotonic':deadline,'pids':{k:p.pid for k,p in processes.items()}})
        while any(p.poll() is None for p in processes.values()):
            if time.monotonic()>=deadline:
                for p in processes.values():
                    if p.poll() is None:p.send_signal(signal.SIGUSR1)
                raise RuntimeError('group deadline reached; current budget canceled, no new window')
            time.sleep(2)
        results=[json.loads((DEST/code/'summary.json').read_text()) for code in ORDER]
        assert sum(r.get('physical_requests',0) for r in results)<=16
        assert sum(r.get('charged_tokens',0) for r in results)<=1100000
        dump(DEST/'group-execution.json',{'executed':sum('business_started_monotonic' in r for r in results),'results':results,
            'source_code_unchanged':code_hashes()==json.loads((DEST/'authorization.json').read_text())['code_sha256'],
            'supervision_wall_seconds':round(time.monotonic()-started,3),'exit_codes':{k:p.returncode for k,p in processes.items()}})
        print('GROUP_DONE '+json.dumps({'results':[{k:r.get(k) for k in ('instrument_id','readable','review_status','physical_requests','charged_tokens','exception')} for r in results]},ensure_ascii=False),flush=True)
    finally:
        for handle in handles.values():handle.close()


if __name__=='__main__':
    parser=argparse.ArgumentParser();parser.add_argument('action',choices=('prepare','run','group'));parser.add_argument('directory',nargs='?');parser.add_argument('code',nargs='?');parser.add_argument('deadline',nargs='?')
    args=parser.parse_args()
    if args.action=='prepare':prepare()
    elif args.action=='group':group(Path(args.directory))
    else:run_one(Path(args.directory),args.code,float(args.deadline))
