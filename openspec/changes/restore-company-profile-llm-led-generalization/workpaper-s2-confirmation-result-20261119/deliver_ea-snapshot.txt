def read(p):return json.loads(p.read_text())
def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()
def dump(p,x):
 p.parent.mkdir(parents=True,exist_ok=True);p.write_text(json.dumps(x,ensure_ascii=False,indent=2,default=str)+'\n')

async def verify_api(reader,saved):
 import httpx
 from fastapi import FastAPI
 from api.company_profile_routes import annual_profile_reader,router
 app=FastAPI();app.include_router(router,prefix='/api/v1');app.dependency_overrides[annual_profile_reader]=lambda:reader
 statuses={};by_id={p['instrument_id']:p for p in saved['profiles']}
 with patch('socket.create_connection',side_effect=AssertionError('external connections forbidden')):
  async with asyncio.timeout(180):
   async with httpx.AsyncClient(transport=httpx.ASGITransport(app=app),base_url='http://local') as client:
    catalog=await client.get('/api/v1/research/company/annual-profiles')
    assert catalog.status_code==200 and catalog.json()==saved['catalog']
    for code in saved['requested_instrument_ids']:
     result=await client.get(f'/api/v1/research/company/{code}/annual-profile')
     assert result.status_code==(200 if code in by_id else 404)
     if code in by_id:assert result.json()==by_id[code]
     statuses[code]=result.status_code
 return dict(catalog_http_status=200,detail_statuses=statuses,equal_read_owner=True,external_connections_forbidden=True,new_model_requests=0)
