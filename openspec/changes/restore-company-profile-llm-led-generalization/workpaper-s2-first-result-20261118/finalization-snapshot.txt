# File: finalize_ds.py
"""6.DS finite delivery of saved real results; no model/network calls."""
from collections import Counter
import hashlib
import json
from pathlib import Path

from research.company_profile.runtime import CompanyProfileResearchWriter, CompanyProfileRuntimeRecordV3
from research.company_profile.reads import CompanyProfileReadService
from research.company_profile.workpaper import workpaper_elements, workpaper_fact_hash
from research.company_profile.workpaper_delivery import bind_workpaper_use_review

ROOT = Path('/home/python/Quote')
DEST = ROOT / 'openspec/changes/restore-company-profile-llm-led-generalization/workpaper-s2-first-result-20261118'
RESEARCH = ROOT / 'data/research/company_profile_s2_first_20261118'


def load(path):
    return json.loads(path.read_text())


def dump(path, payload):
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(payload, ensure_ascii=False, indent=2) + '\n')


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def main():
    audit_path = DEST / 'audit-decisions.json'
    audit = load(audit_path)
    group = load(DEST / 'group-execution.json')
    manifest = load(DEST / 'source-manifest.json')['companies']
    assert len(manifest) == group['executed'] == 20 and not group['not_started']
    writer = CompanyProfileResearchWriter(RESEARCH)
    rows, budgets, checked, original_hashes = [], [], [], {}
    total = Counter()
    all_api = []
    for item in manifest:
        code = item['instrument_id']
        folder = DEST / code[:6]
        summary = load(folder / 'summary.json')
        budget_path = folder / 'budget.json'
        budget = load(budget_path) if budget_path.exists() else {'physical_requests': 0, 'charged_tokens': 0, 'attempts': []}
        attempts = budget['attempts']
        assert budget['physical_requests'] == len(attempts) <= 10
        assert budget['charged_tokens'] == sum(a['charged_tokens'] for a in attempts) <= 1000000
        counts = Counter(a['logical_request_id'] for a in attempts)
        assert max(counts.values(), default=0) <= 2
        recoveries = sum(n - 1 for n in counts.values())
        assert recoveries <= 3
        unknown, usage = 0, 0
        for attempt in attempts:
            known = (attempt.get('usage') or {}).get('total_tokens')
            if known is None:
                unknown += attempt['charged_tokens']
            else:
                assert attempt['charged_tokens'] == known
                usage += known
        total.update(physical_requests=len(attempts), charged_tokens=budget['charged_tokens'],
                     recoveries=recoveries, known_supplier_total=usage, unknown_usage_reservations=unknown)
        budgets.append({'instrument_id': code, 'physical_requests': len(attempts), 'recoveries': recoveries,
                        'ledger': budget['charged_tokens'], 'known_supplier_total': usage,
                        'unknown_usage_reservations': unknown, 'fee': 'unknown'})
        row = {'instrument_id': code, 'name': item['name'], 'raw_fact_count': 0,
               'status': 'no_profile', 'independent_core_answers': None,
               'business_seconds': summary.get('business_seconds'), 'review_status': summary.get('review_status')}
        path = folder / 'runtime.json'
        if path.exists():
            raw = path.read_bytes()
            original_hashes[str(path.relative_to(ROOT))] = hashlib.sha256(raw).hexdigest()
            record = CompanyProfileRuntimeRecordV3.model_validate_json(raw)
            original = record.workpaper.model_dump(mode='json')
            facts = {f.fact_id: f for e in workpaper_elements(record.workpaper).values() for f in e.facts}
            assert len(facts) == summary['total_facts_with_optional']
            row.update(raw_fact_count=len(facts), status='content_requires_remaining_independent_assessment')
            decisions = []
            company = audit['companies'].get(code)
            if company:
                ids = company['checked_fact_ids'] + company.get('additional_checked_fact_ids', [])
                assert len(ids) == len(set(ids)) and set(ids) <= set(facts)
                row['independent_core_answers'] = company['core']
                row['status'] = ('basic_description_source_supported' if all(company['core'].values())
                                 else 'basic_description_incomplete')
                for fid in ids:
                    fact = facts[fid]
                    reason = company['basis']
                    disposition = 'source_checked'
                    if fid in company.get('errors', {}):
                        disposition, reason = 'withheld', company['errors'][fid]
                    elif fid in company.get('duplicates', {}):
                        disposition, reason = 'duplicate', company['duplicates'][fid]
                    limitation = company.get('limitations', {}).get(fid)
                    provenance = company.get('provenance_limitations', {})
                    if provenance:
                        limitation = '；'.join(filter(None, [limitation, *provenance.values()]))
                    decisions.append({'fact_id': fid, 'fact_sha256': workpaper_fact_hash(fact),
                                      'disposition': disposition, 'evidence_ids': list(fact.evidence_ids),
                                      'reason': reason, 'limitation': limitation})
                    checked.append({'instrument_id': code, **decisions[-1]})
                paper = bind_workpaper_use_review(record.workpaper, {
                    'report': record.report.model_dump(mode='json'), 'reviewer': 'Codex independent source audit 6.DS',
                    'reference': str(audit_path.relative_to(ROOT)), 'reference_sha256': sha(audit_path), 'decisions': decisions})
                candidate = paper.model_dump(mode='json')
                candidate.pop('use_review', None)
                prior = dict(original)
                prior.pop('use_review', None)
                assert candidate == prior, 'only local use-review may change'
                record = record.model_copy(update={'workpaper': paper})
            target = writer.output_root / f'{record.work_id}.json'
            if target.exists():
                assert target.read_text() == record.model_dump_json(indent=2), 'do not overwrite unrelated output'
            else:
                writer.persist(record)
            total['raw_facts'] += len(facts)
            total['profiles_with_content'] += 1
        rows.append(row)
    assert total['physical_requests'] == 88 and total['charged_tokens'] == 2532366
    assert total['raw_facts'] == 475 and total['profiles_with_content'] == 16
    assert len(checked) == 227
    instruments = [i['instrument_id'] for i in manifest]
    reader = CompanyProfileReadService(RESEARCH)
    query = reader.query(instruments)
    assert query['delivered'] == 16
    assert set(query['missing_instrument_ids']) == {'600113.SH', '000025.SZ', '000001.SZ', '000166.SZ'}
    exported = reader.export(instruments, export_directory=DEST / 'delivery')
    assert query['profiles'] == exported['profiles']
    assert CompanyProfileReadService(RESEARCH).query(instruments)['profiles'] == query['profiles']
    delivery_counts = Counter()
    for row in rows:
        if row['status'] == 'no_profile':
            continue
        api = reader.research_profile(row['instrument_id'])
        assert api == CompanyProfileReadService(RESEARCH).research_profile(row['instrument_id'])
        visible = [entry for entry in api['facts'] if entry['fact'] is not None]
        row['visible_fact_count'] = len(visible)
        row['visible_metric_count'] = sum(len(entry['fact']['metrics']) for entry in visible)
        row['gap_count'] = len(api['information_gaps'])
        row['suggested_acquisition_count'] = len(api['suggested_acquisitions'])
        for entry in api['facts']:
            delivery_counts[entry['state']] += 1
        row['independently_checked_objects'] = sum(entry['review_basis'] == 'independent_source_check' for entry in api['facts'])
        total['visible_facts'] += row['visible_fact_count']
        total['visible_metrics'] += row['visible_metric_count']
        all_api.append(api)
    for name, digest in original_hashes.items():
        assert sha(ROOT / name) == digest
    dump(DEST / 'resource-audit.json', {'per_company': budgets, 'totals': dict(total),
        'group_supervision_wall_seconds': group['supervision_wall_seconds'], 'fee': 'unknown',
        'supplier_total_not_input_plus_output': True, 'preparation_and_independent_review_time': 'not_completely_measured'})
    dump(DEST / 'delivery' / 'annual-api.json', {'catalog': reader.research_profile_catalog(),
        'profiles': all_api, 'requested_instrument_ids': instruments, 'missing_instrument_ids': query['missing_instrument_ids']})
    dump(DEST / 'batch-assessment.json', {'phase': 'S2_first_group', 'status': 'not_passed', 'executed': 20,
        'not_started': [], 'nonempty_profiles': 16, 'basic_complete_candidate_upper_bound': 14,
        'required_first_delivery_companies': 18, 'basic_recall_upper_bound': 0.75,
        'required_basic_macro_recall': 0.95, 'full_group_audited_accuracy': 'not_assessed',
        'required_audited_accuracy': 0.99, 'independent_objects_checked': len(checked),
        'independent_companies_checked': len(audit['companies']), 'known_independent_errors_withheld': 1,
        'local_duplicate_not_redisplayed': 1, 'full_formal_400_supplement_sample': 'not_completed',
        'delivery_states': dict(delivery_counts), 'research_output_root': str(RESEARCH), 'rows': rows,
        'limits': ['75%为由4无稿及2实质不完整推得的最有利上限，不是已完成整组召回计分。',
                   '14家只是其余全部成立时的上限，不是已独立通过14家。',
                   '227对象为有限后验核查，不推算全组准确率；未独立核对象保实际模型状态。',
                   'S1旧失败不改，S2本组不通过，未授权确认组、S3或生产。']})
    dump(DEST / 'readback-check.json', {'query_export_new_reader_equal': True, 'api_detail_fresh_reader_equal': True,
        'profile_count': 16, 'missing_instrument_ids': query['missing_instrument_ids'], 'raw_runtime_sha256_unchanged': original_hashes,
        'review_reference_sha256': sha(audit_path), 'new_model_requests_in_finalization': 0})
    print(json.dumps({'totals': dict(total), 'delivery_states': dict(delivery_counts), 'root': str(RESEARCH)}, ensure_ascii=False))


if __name__ == '__main__':
    main()


# File: verify_api.py
"""Read-only in-process ASGI verification; no model or TCP requests."""
import asyncio
import json
from pathlib import Path
from unittest.mock import patch

import httpx
from fastapi import FastAPI
from api.company_profile_routes import annual_profile_reader, router
from research.company_profile.reads import CompanyProfileReadService

ROOT = Path('/home/python/Quote')
DEST = ROOT / 'openspec/changes/restore-company-profile-llm-led-generalization/workpaper-s2-first-result-20261118'
reader = CompanyProfileReadService(ROOT / 'data/research/company_profile_s2_first_20261118')


async def main():
    saved = json.loads((DEST / 'delivery/annual-api.json').read_text())
    app = FastAPI()
    app.include_router(router, prefix='/api/v1')
    app.dependency_overrides[annual_profile_reader] = lambda: reader
    statuses = {}
    with patch('socket.create_connection', side_effect=AssertionError('external sockets forbidden')):
        async with asyncio.timeout(120):
            async with httpx.AsyncClient(transport=httpx.ASGITransport(app=app), base_url='http://local') as client:
                catalog = await client.get('/api/v1/research/company/annual-profiles')
                assert catalog.status_code == 200 and catalog.json() == saved['catalog']
                by_code = {p['instrument_id']: p for p in saved['profiles']}
                for code in saved['requested_instrument_ids']:
                    response = await client.get(f'/api/v1/research/company/{code}/annual-profile')
                    expected = 200 if code in by_code else 404
                    assert response.status_code == expected
                    if expected == 200:
                        assert response.json() == by_code[code]
                    statuses[code] = response.status_code
    result = {'catalog_http_status': 200, 'detail_statuses': statuses,
              'details_equal_canonical_api': True, 'external_sockets_forbidden': True,
              'new_model_requests': 0, 'runtime_default_configuration_changed': False}
    (DEST / 'api-check.json').write_text(json.dumps(result, ensure_ascii=False, indent=2) + '\n')
    print(json.dumps(result, ensure_ascii=False), flush=True)


if __name__ == '__main__':
    asyncio.run(main())
