def boundary_check():
    from research.company_profile.stage5_provider import CommonGatewaySemanticProvider
    from research.company_profile.models import ReportIdentity
    from research.company_profile.workpaper import WorkpaperReviewResponse,WorkpaperRepairResponse,WorkpaperRepairVerificationResponse
    calls=[n for n in ast.walk(ast.parse(source)) if isinstance(n,ast.Call) and isinstance(n.func,ast.Name) and n.func.id=='CommonGatewaySemanticProvider']
    assert len(calls)==1
    cap=next(ast.literal_eval(k.value) for k in calls[0].keywords if k.arg=='timeout_seconds')
    assert cap==2450
    config=config_for_run(); asset=source_items()[0]['asset']
    report=ReportIdentity(instrument_id=asset['instrument_id'],report_id=asset['filing_id'],document_version=asset['content_hash'],report_period=asset['report_period'],published_at=asset['published_at'])
    captured=[]
    class Boundary(BaseException):pass
    class Client:
        physical_budget=object()
        def __init__(self):self.config=config
        async def complete(self,request):
            captured.append({'purpose':request.metadata.get('purpose'),'profile':request.profile,'timeout_seconds':request.timeout_seconds})
            raise Boundary()
    for purpose,model,profile in [('review_profile_workpaper',WorkpaperReviewResponse,REVIEW),('repair_profile_workpaper',WorkpaperRepairResponse,EXTRACT),('verify_profile_workpaper_repair',WorkpaperRepairVerificationResponse,REVIEW)]:
        provider=CommonGatewaySemanticProvider(client=Client(),profile=EXTRACT,max_output_tokens=16384,timeout_seconds=cap,execution_authorization={'status':'authorized','document_version':report.document_version,'approval_reference':APPROVAL})
        try:
            provider.workpaper_request(request_id='ds-bind:'+purpose,purpose=purpose,report=report,payload={'source_text':'Local parameter boundary; no model call'},instructions='Local boundary only',response_model=model,max_output_tokens=16384,timeout_seconds=2450,profile_override=profile)
        except Boundary:pass
    assert [r['timeout_seconds'] for r in captured]==[2450,2450,2450]
    assert config.profiles[REVIEW].attempt_timeout_seconds==1200
    assert config.profiles[EXTRACT].attempt_timeout_seconds==600
    dump(DEST/'binding-check.json',{'provider_cap':cap,'gateway_boundaries':captured,'profile_physical_seconds':{'review':1200,'flash':600},'new_model_requests':0})
    print('BINDING_CHECK_OK',flush=True)
