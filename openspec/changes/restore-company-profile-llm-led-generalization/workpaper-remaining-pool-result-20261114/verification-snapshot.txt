"""Read-only verification of the four completed BX artifacts; no model access."""
import hashlib
import importlib.util
import json
from pathlib import Path
import sys

ROOT = Path('/home/python/Quote')
sys.path.insert(0, str(ROOT))
CHANGE = ROOT / 'openspec/changes/restore-company-profile-llm-led-generalization'
from research.company_profile.workpaper import CompanyProfileWorkpaper, WorkpaperDraftResponse, bind_workpaper_draft, normalize_workpaper_draft
from research.company_profile.workpaper_review import workpaper_hash

def sha(path): return hashlib.sha256(path.read_bytes()).hexdigest()
def facts(paper):
    parts = [*paper.elements.__dict__.values(), *(paper.supply_chain.__dict__.values() if paper.supply_chain else [])]
    return {f.fact_id: f.model_dump(mode='json') for e in parts for f in e.facts}

spec = importlib.util.spec_from_file_location('bx_operator', ROOT / 'data/QuoteBak/workpaper_bx_run.py')
operator = importlib.util.module_from_spec(spec)
spec.loader.exec_module(operator)
frozen = json.loads((operator.DEST / 'authorization.json').read_text())
assert operator.code_hashes() == frozen['code_sha256']
assert sha(ROOT / 'config/13_llm.json') == frozen['shared_config_sha256']
assert sha(ROOT / 'data/QuoteBak/workpaper_bx_run.py') == frozen['operator_sha256']
results = []
for symbol in ('600082', '600097', '600054', '600000'):
    run = CHANGE / 'workpaper-runs' / ('v10-remaining-pool-' + symbol + '-20261114')
    query = json.loads((run / 'query.json').read_text())
    paper = CompanyProfileWorkpaper.model_validate(query['profiles'][0]['workpaper'])
    receipts = json.loads((run / 'model-receipts.json').read_text())
    for receipt in receipts:
        assert hashlib.sha256(receipt['raw_content'].encode()).hexdigest() == receipt['response_hash']
        assert json.loads(receipt['raw_content']) == receipt['response_payload']
    raw = next(r['response_payload'] for r in receipts if r['purpose'] == 'extract_profile_workpaper')
    seed = bind_workpaper_draft(WorkpaperDraftResponse.model_validate(normalize_workpaper_draft(raw)),
        report=paper.report, reading_scope=paper.reading_scope, evidence=paper.evidence, generation=paper.generation)
    assert workpaper_hash(seed) == paper.quality_review.original_draft_hash
    before, after = facts(seed), facts(paper)
    adopted = {f.affected_fact_id or 'review-add:' + f.finding_id: f.revised_fact.model_dump(mode='json')
               for f in paper.quality_review.findings if f.resolution in ('corrected_pending_trust', 'added_pending_trust')}
    for key, value in after.items():
        assert value == (adopted[key] if key in adopted else before[key])
    removed = set(before) - set(after)
    assert removed <= {f.affected_fact_id for f in paper.quality_review.findings}
    budget = json.loads((run / 'budget.json').read_text())
    counts = {}
    for attempt in budget['attempts']:
        key = attempt.get('logical_request_id') or attempt['request_id']
        counts[key] = counts.get(key, 0) + 1
        assert attempt['status'] == 'success' and attempt['finish_reason'] == 'stop'
        assert attempt['source_label'] in {'deepseek:deepseek-flash', 'scorpio:grok-4.7'}
    assert budget['physical_requests'] <= 10 and budget['charged_tokens'] <= 500000
    assert sum(n - 1 for n in counts.values()) <= 3 and all(n <= 2 for n in counts.values())
    assert not paper.quality_review.retained_request_ids and not paper.quality_review.locally_revalidated_request_ids
    summary = json.loads((run / 'summary.json').read_text())
    assert summary['three_reads_equal'] and summary['business_seconds'] <= 1800
    assert query['profiles'][0]['core_complete'] is False
    binding = json.loads((run / 'binding.json').read_text())
    assert sha(Path(binding['asset']['archive_path'])) == paper.report.document_version
    results.append({'instrument_id': paper.report.instrument_id, 'fact_count': len(after),
        'metric_count': query['profiles'][0]['workpaper_summary']['metric_count'],
        'independent_facts_unchanged': True, 'adopted_fact_ids': sorted(adopted),
        'removed_fact_ids': sorted(removed), 'original_draft_hash_matches': True,
        'raw_hashes_and_json_match': True, 'retained_receipts_loaded': False,
        'three_reads_equal': True, 'query_sha256': sha(run / 'query.json'),
        'physical_requests': budget['physical_requests'], 'charged_tokens': budget['charged_tokens']})
assert sum(r['physical_requests'] for r in results) <= 40
assert sum(r['charged_tokens'] for r in results) <= 2000000
group = json.loads((operator.DEST / 'group-execution.json').read_text())
assert group['business_wall_seconds'] <= 2400 and group['source_code_unchanged']
print('VALIDATION_JSON=' + json.dumps({'task': '6.BX', 'verification_mode': 'read_only_no_new_calls',
    'actual_frozen_code_and_config_unchanged': True, 'results': results,
    'physical_requests': sum(r['physical_requests'] for r in results),
    'actual_usage': sum(r['charged_tokens'] for r in results), 'recoveries': 0,
    'cost': 'unknown', 'production_authorization': 'not_authorized'}, ensure_ascii=False))
