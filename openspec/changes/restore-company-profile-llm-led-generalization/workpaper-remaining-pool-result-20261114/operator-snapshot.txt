def code_hashes():
    paths=sorted((ROOT/'research/company_profile').glob('*.py'))+sorted((ROOT/'utils/llm').glob('*.py'))
    paths += [ROOT/'research/storage.py',ROOT/'research/business_profile_pdf_artifacts.py',ROOT/'research/business_profile_production_operations.py']
    return {str(p.relative_to(ROOT)):sha(p) for p in paths}

def dump(path, value):
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2, default=str) + '\n')

def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def source_items():
    items = [i for i in json.loads(MANIFEST.read_text())['companies'] if i['instrument_id'] in {'600082.SH','600097.SH','600054.SH','600000.SH'}]
    assert [i['instrument_id'] for i in items] == ['600082.SH','600097.SH','600054.SH','600000.SH']
    return items

def prepare_one(item, directory):
    from research.business_profile_pdf_artifacts import ensure_archived_pdf_page_artifact
    from research.company_profile.workpaper_sections import index_sections
    started=time.monotonic(); asset=item['asset']; symbol=item['instrument_id'][:6]
    assert sha(Path(asset['archive_path'])) == asset['content_hash']
    parsed=ensure_archived_pdf_page_artifact(asset)
    artifact=parsed['artifact']
    pages=[{'page':p.page_number,'text':p.text} for p in artifact.pages]
    assert len(pages)==item['pdf_pages']
    index=index_sections(pages)
    prepared={'asset':asset,'name':item['name'],'pages':pages,
              'index':[s.model_dump(mode='json') for s in index],
              'preparation_seconds':round(time.monotonic()-started,3),
              'source_preparation':{k:parsed.get(k) for k in ('artifact_path','artifact_hash','status','cache_status','timings')}}
    dump(directory/(symbol+'.json'),prepared)
    print(json.dumps({'prepared':symbol,'pages':len(pages),'source_seconds':prepared['preparation_seconds'],'cache_status':parsed.get('cache_status')},ensure_ascii=False),flush=True)
    return symbol

def prepare():

    directory=Path(tempfile.mkdtemp(prefix='workpaper-bx-prepared-'))
    print('PREPARED_ROOT='+str(directory),flush=True)
    with ThreadPoolExecutor(max_workers=4) as pool:
        list(pool.map(lambda item:prepare_one(item,directory),source_items()))
    print('PREPARE_DONE='+str(directory),flush=True)

def inspect(directory, symbol, pages=None, sections=None):
    prepared=json.loads((directory/(symbol+'.json')).read_text())
    if sections:
        for entry in prepared['index']:
            if re.search(sections,entry['section_path'][-1]):
                print(json.dumps({k:entry[k] for k in ('section_id','section_path','start_page','end_page','characters')},ensure_ascii=False))
        return
    from research.company_profile.workpaper_sections import extract_selected_layout
    page_numbers=[int(v) for v in pages.split(',')]
    for page in extract_selected_layout(Path(prepared['asset']['archive_path']).read_bytes(), page_numbers):
        print('\nPHYSICAL PAGE',symbol,page['page'])
        print(page['text'])

def group(directory):
    # One batch only. A signal cancels the existing per-company budget before
    # further sends; an in-flight response is still settled/persisted normally.
    group_start=time.monotonic(); deadline=group_start+2400
    items=source_items(); processes={}; logs={}; results={}; stop=None
    for item in items:
        symbol=item['instrument_id'][:6]
        log=directory/(symbol+'-run.log'); logs[symbol]=log.open('w')
        processes[symbol]=subprocess.Popen([sys.executable,str(Path(__file__)),'run',
            '--prepared',str(directory),'--symbol',symbol,'--group-deadline',str(deadline)],
            stdout=logs[symbol],stderr=subprocess.STDOUT)
    while len(results)<len(items):
        for symbol,process in processes.items():
            if symbol in results or process.poll() is None:
                continue
            logs[symbol].close()
            result_path=CHANGE/'workpaper-runs'/('v10-remaining-pool-'+symbol+'-20261114')/'summary.json'
            results[symbol]=json.loads(result_path.read_text()) if result_path.exists() else {
                'instrument_id':symbol+'.SH','exception':'operator_exit_'+str(process.returncode),'readable':False}
            print('COMPANY_DONE '+json.dumps(results[symbol],ensure_ascii=False),flush=True)
        ordered=[results.get(i['instrument_id'][:6]) for i in items]
        if stop is None:
            if time.monotonic()>=deadline:
                stop='group_deadline'
            else:
                for left,right in zip(ordered,ordered[1:]):
                    if left is None or right is None: continue
                    x=left.get('failure_signature') or left.get('exception')
                    y=right.get('failure_signature') or right.get('exception')
                    if x and x==y:
                        stop='consecutive_same_core_failure: '+x
                        break
            if stop:
                dump(DEST/'group-stop.json',{'reason':stop,'at_monotonic':time.monotonic()})
                for process in processes.values():
                    if process.poll() is None: process.send_signal(signal.SIGUSR1)
        # Also permit the supervising source audit to cancel later sends.
        if (DEST/'independent-stop.json').exists() and stop is None:
            stop='independent_source_stop'
            for process in processes.values():
                if process.poll() is None: process.send_signal(signal.SIGUSR1)
        dump(DEST/'progress.json',{'executed':len(results),'results':[v for v in ordered if v],
            'stop_reason':stop,'supervision_wall_seconds':round(time.monotonic()-group_start,3)})
        if len(results)<len(items): time.sleep(1)
    ordered=[results[i['instrument_id'][:6]] for i in items]
    starts=[r['business_started_monotonic'] for r in ordered if r.get('business_started_monotonic')]
    ends=[r['business_ended_monotonic'] for r in ordered if r.get('business_ended_monotonic')]
    dump(DEST/'group-execution.json',{'executed':len(results),'not_started':[],
        'results':ordered,'stop_reason':stop,'supervision_wall_seconds':round(time.monotonic()-group_start,3),
        'business_wall_seconds':round(max(ends)-min(starts),3) if starts and ends else None,
        'source_code_unchanged':code_hashes()==json.loads((DEST/'authorization.json').read_text())['code_sha256']})
    print('GROUP_DONE '+str(len(results))+' '+str(stop),flush=True)

def main():
    parser=argparse.ArgumentParser(); parser.add_argument('mode',choices=['prepare','inspect','freeze','run','group'])
    parser.add_argument('--prepared',type=Path); parser.add_argument('--symbol'); parser.add_argument('--pages'); parser.add_argument('--sections')
    parser.add_argument('--group-deadline',type=float)
    args=parser.parse_args()
    if args.mode=='prepare': prepare()
    elif args.mode=='inspect': inspect(args.prepared,args.symbol,args.pages,args.sections)
    elif args.mode=='freeze': freeze(args.prepared)
    elif args.mode=='run': run_one(args.prepared,args.symbol,args.group_deadline)
    else: group(args.prepared)

if __name__=='__main__':main()
