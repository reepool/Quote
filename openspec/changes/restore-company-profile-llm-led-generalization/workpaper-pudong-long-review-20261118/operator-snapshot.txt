def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()

def read(path):
    return json.loads(Path(path).read_text())

def dump(path, value):
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2, default=str)+'\n')

    return {f.fact_id: f for e in workpaper_elements(paper).values() for f in e.facts}

def bind_controls(paper,plan):
    from research.company_profile.workpaper import workpaper_fact_hash
    from research.company_profile.workpaper_delivery import bind_workpaper_use_review
    controls={d['fact_id']:d for d in plan['source_independent_controls']['decisions']}
    decisions=[]
    for fid,fact in facts(paper).items():
        old=controls.get(fid)
        if old is None:
            continue
        digest=workpaper_fact_hash(fact)
        if digest==old['fact_sha256']:
            decisions.append(old)
        elif old['disposition'] in {'withheld','source_only','duplicate'}:
            decisions.append({'fact_id':fid,'fact_sha256':digest,'disposition':'source_only',
                'evidence_ids':list(fact.evidence_ids),'reason':'原受限对象已修改，须另核原文；模型接受不自动解除独立限制。'})
    return bind_workpaper_use_review(paper,{'report':paper.report.model_dump(mode='json'),
        'reviewer':'Codex继承原独立核源与两项范围限制，未核全部输出',
        'reference':str(PLAN),'reference_sha256':sha(PLAN),'decisions':decisions})
