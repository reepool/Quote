import hashlib
import json
from pathlib import Path
from pypdf import PdfReader

C = Path('openspec/changes/deliver-company-profile-m4-v45-unseen-small-batch')
frozen = json.loads((C / 'independent-frozen-source-pages.json').read_text())
supplemental = json.loads((C / 'actual-object-supplemental-source-pages.json').read_text())
out = {}
for instrument, data in frozen.items():
    pdf = Path(data['pdf'])
    assert hashlib.sha256(pdf.read_bytes()).hexdigest() == data['document_version']
    reader = PdfReader(pdf)
    pages = supplemental[instrument]['pages']
    for page in pages:
        assert reader.pages[page['page'] - 1].extract_text() == page['text']
    out[instrument] = {
        'document_version': data['document_version'],
        'additional_actual_evidence_pages': [p['page'] for p in pages],
        'all_additional_texts_equal_official': True,
        'total_distinct_verified_pages': len(data['pages']) + len(pages),
    }
(C / 'actual-evidence-official-source-validation.json').write_text(json.dumps({
    'instruments': out,
    'frozen_plain_page_count': sum(len(v['pages']) for v in frozen.values()),
    'additional_plain_page_count': sum(len(v['pages']) for v in supplemental.values()),
    'total_distinct_verified_pages': sum(v['total_distinct_verified_pages'] for v in out.values()),
    'count_note': '执行前82不同页，实际输出补充页独立核验，不改原合同和冻结页。',
}, ensure_ascii=False, indent=2) + '\n')
print('EXTRA_SOURCES_VERIFIED', sum(len(v['pages']) for v in supplemental.values()))
