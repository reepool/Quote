 rows=scope['source_rows'][ins];roles=[r for r in scope['native_roles'] if r['instrument_id']==ins];pages={p['page']:p for p in fixture[ins]['independent_pages']};facts={};quotes=[]
 def normalize(t):
  t=compact(t)
  return re.sub(r'\d{1,3}/(?:185|218)','',re.sub(r'(?:黄山旅游发展股份有限公司|中信尼雅葡萄酒股份有限公司)2025年年度报告','',t))
 for idx,r in enumerate(raw):

  assert r['reported_period']=='2025-12-31' and r['report']['document_version']==scope['versions'][ins]['sha256']
  if kind in ['Segment','Measurement']:
   matches=[j for j,row in enumerate(rows) if native_match(r,row)]
   assert num(n['value']) in num(source),(ins,idx,n)

    assert ins=='600084.SH' and n['name']=='正常经营之外的其他业务收入' and n['value']=='555.89' and n['unit']=='万元' and n['header']=='其他业务收入扣除明细/本年度' and r['subject_scope']=='consolidated_group' and 10 in ep,(ins,idx,n)

   for token in re.findall(r'\d[\d,]*(?:\.\d+)?',r['source_text']):

   def ordered(a,b):

   supported=txt in src or ordered(txt,src);mode='plain_original_fragments_with_page_headers_removed'

    layouts=''.join(compact(pages[p].get('layout_text','')) for p in ep);lay=normalize(layouts);supported=txt in lay or ordered(txt,lay);mode='independent_layout_reading_order'

   if not supported:raise AssertionError((ins,idx,'unsupported overview',r['source_text'][:80],ep))
   d.update(accurate=True,reason='独立官方原段/表格及必要续页支持；表引用保留上期/不适用/原主体，不作为当前肯定收入。',source_quote_supported=True,original_reading_mode=mode)
   quotes.append(dict(record_id=r['record_id'],pages=ep,reading_mode=mode))
  elif kind in ['Activity','Relationship']:
   matches=[j for j,role in enumerate(roles) if action_matches([r],role)]
   if matches:
    # Stronger than the original aliases: three platform objects remain distinct.
    assert len(matches)==1,(ins,idx,matches)
    d.update(accurate=True,reason='原具名对象、动作和原主体绑定成立；生产/采购/销售互不推导，状态限定按原句核。',recall_role_index=matches[0],matched_source_condition_id=roles[matches[0]]['condition_id'])
   elif ins=='600054.SH' and n['name'] in ['旅游产品','活动策划']:
