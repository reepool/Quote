def audit(directory):

 for i in scope['revenue_conditions']:

   actual_signature={x:r.get(x)for x in FIELDS}
   explicit=decisions.get(i+'|'+r['record_id'])
   if not explicit:
    # Fixture and formal report identities produce different Evidence IDs.
    # Bind only an exact native semantic snapshot AND identical official pages.
    candidates=[d for key,d in decisions.items()if key.startswith(i+'|')and d['expected_semantics']==actual_signature and d['source_pages']==sorted(pages(r))]

   ok=bool(hit or roles or same)
   reason='原收入行/栏目/主体/交易方/期间/单位/本期取格匹配'if hit else'原具名对象/动作方向/分类/主体/期间匹配'if roles else explicit['reason']if same else'未有原表匹配或与逐对象原文复核快照不同，不默认准确'
   objects.append(dict(instrument_id=i,object_id=r['record_id'],object_type=k,record=r,decision=dict(accurate=ok,critical_numeric_error=False,matched_income_conditions=hit,matched_role_conditions=roles,explicit_source_decision=same,reason=reason)))
  for v in scope['revenue_conditions'][i]:
   ids={kind:[r['record_id']for r in rs if r['object_type']==kind and match(r,v)]for kind in ['Segment','Measurement']};mapping.append(dict(condition_id=v['condition_id'],objects=ids));
   if not all(ids.values()):missing.append(v['condition_id'])
  for v in scope['action_conditions']:

   ids=[r['record_id']for r in rs if r['object_type']=='Activity' and rolematch(r,v)];mapping.append(dict(condition_id=v['condition_id'],objects=ids))
   if not ids:missing.append(v['condition_id'])

   support=set(a['supporting_record_ids']);known={r['record_id']for r in rs};ok=bool(a['answered'])and all(x['actual_complete']for x in obligations)and support<=known

  ng.extend(negatives(i,rs,profile))
 # Every source condition needs a distinct native object; a broad service enum
 # must not let one installation record also satisfy maintenance or support.
 assigned={}
 def assign(condition,seen):
  candidates=condition['objects']
  if isinstance(candidates,dict):candidates=candidates['Segment']
  for record_id in candidates:
   if record_id in seen:continue
   seen.add(record_id)
   prior=assigned.get(record_id)
   if prior is None or assign(prior,seen):
    assigned[record_id]=condition
    return True
  return False
