import faulthandler,json,sqlite3,time
from pathlib import Path
start=time.monotonic()
def mark(event,**data):print(json.dumps(dict(event=event,elapsed=time.monotonic()-start,**data),ensure_ascii=False),flush=True)

original_connect=sqlite3.connect
last_progress=0
sql_count=0
class ReadOnlyConnection(sqlite3.Connection):
 def __init__(self,*args,**kwargs):
  super().__init__(*args,**kwargs)
  self.set_trace_callback(self.trace)
  self.set_progress_handler(self.progress,100000)
 def trace(self,sql):
  global sql_count
  sql_count+=1;self.last_sql=sql
 def progress(self):
  global last_progress
  now=time.monotonic()
  if now-last_progress>2:
   mark('sql_progress',sql=self.last_sql,sql_count=sql_count);last_progress=now
  return 0
def readonly_connect(database,*args,**kwargs):
 db=str(database)
 if db!=':memory:':db=Path(db).resolve().as_uri()+'?mode=ro'
 kwargs.update(uri=True,factory=ReadOnlyConnection)
 return original_connect(db,*args,**kwargs)
sqlite3.connect=readonly_connect
