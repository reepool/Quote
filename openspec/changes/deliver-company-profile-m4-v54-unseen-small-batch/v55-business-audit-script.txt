def evaluate(ins,rs,profile):

 for a in profile['dimensions']:

  answers[a['dimension_id']]={'accurate':a['answered'] and bool(a['excerpt']) and not bad and not supports and digest(a['excerpt'])==manual['reviewed_answers'][ins+':'+a['dimension_id']]['excerpt_sha256'],'critical_numeric_error':False,'reason':'独立正文实质核：完整业务链、产品矩阵及收费/交付确认机制，与原母子/参股/报告日/筹备/研发和适用状态贯通；九项冲突单位不进入确定性答案','failed_substance_groups':bad,'inaccurate_supporting_ids':supports,'source_substance':rule['substance']}
